    61→// Decode double-encoded HTML entities
    62→function decodeHtmlEntities(text: string): string {
    63→  if (!text) return '';
    64→  let decoded = text;
    65→  for (let i = 0; i < 3; i++) {
    66→    const prev = decoded;
    67→    decoded = decoded
    68→      .replace(/&amp;/g, '&')
    69→      .replace(/&lt;/g, '<')
    70→      .replace(/&gt;/g, '>')
    71→      .replace(/&quot;/g, '"')
    72→      .replace(/&#39;/g, "'")
    73→      .replace(/&nbsp;/g, ' ');
    74→    if (decoded === prev) break;
    75→  }
    76→  return decoded;
    77→}

    79→// Parse YouTube URL to get embed URL
    80→function getYouTubeEmbedUrl(url: string): string | null {
    81→  try {
    82→    const urlObj = new URL(url);
    83→    // Standard youtube.com/watch?v=...
    84→    if (urlObj.hostname.includes('youtube.com') && urlObj.searchParams.get('v')) {
    85→      return `https://www.youtube.com/embed/${urlObj.searchParams.get('v')}`;
    86→    }
    87→    // youtu.be/...
    88→    if (urlObj.hostname === 'youtu.be') {
    89→      return `https://www.youtube.com/embed/${urlObj.pathname.slice(1)}`;
    90→    }
    91→    // youtube.com/embed/...
    92→    if (urlObj.hostname.includes('youtube.com') && urlObj.pathname.startsWith('/embed/')) {
    93→      return url;
    94→    }
    95→    // Vimeo
    96→    if (urlObj.hostname.includes('vimeo.com')) {
    97→      const videoId = urlObj.pathname.split('/').filter(Boolean).pop();
    98→      return videoId ? `https://player.vimeo.com/video/${videoId}` : null;
    99→    }
   100→    return null;
   101→  } catch {
   102→    return null;
   103→  }
   104→}

   106→function sanitizeHtml(html: string): string {
   107→  if (!html) return '';
   108→  let sanitized = decodeHtmlEntities(html);
   109→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
   110→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
   111→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
   112→  return sanitized;
   113→}

   131→// Star Rating component
   132→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
   133→  const stars = [];
   134→  const fullStars = Math.floor(rating);
   135→  const hasHalf = rating - fullStars >= 0.5;
   136→  for (let i = 0; i < 5; i++) {
   137→    if (i < fullStars) stars.push('full');
   138→    else if (i === fullStars && hasHalf) stars.push('half');
   139→    else stars.push('empty');
   140→  }
   141→  return (
   142→    <div className="flex items-center gap-1">
   143→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
   144→        {stars.map((type, i) => (
   145→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-600'}`} />
   146→        ))}
   147→      </div>
   148→      {count !== undefined && (
   149→        <span className={`text-gray-500 dark:text-gray-300 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
   150→      )}
   151→    </div>
   152→  );
   153→}
