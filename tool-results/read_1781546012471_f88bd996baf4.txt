    49→// Decode double-encoded HTML entities
    50→function decodeHtmlEntities(text: string): string {
    51→  if (!text) return '';
    52→  let decoded = text;
    53→  for (let i = 0; i < 3; i++) {
    54→    const prev = decoded;
    55→    decoded = decoded
    56→      .replace(/&amp;/g, '&')
    57→      .replace(/&lt;/g, '<')
    58→      .replace(/&gt;/g, '>')
    59→      .replace(/&quot;/g, '"')
    60→      .replace(/&#39;/g, "'")
    61→      .replace(/&nbsp;/g, ' ');
    62→    if (decoded === prev) break;
    63→  }
    64→  return decoded;
    65→}
    66→
    67→function sanitizeHtml(html: string): string {
    68→  if (!html) return '';
    69→  let sanitized = decodeHtmlEntities(html);
    70→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
    71→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
    72→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
    73→  return sanitized;
    74→}
