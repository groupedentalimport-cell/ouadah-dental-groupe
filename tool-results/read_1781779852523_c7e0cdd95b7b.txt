    62→// Decode double-encoded HTML entities
    63→function decodeHtmlEntities(text: string): string {
    64→  if (!text) return '';
    65→  let decoded = text;
    66→  for (let i = 0; i < 3; i++) {
    67→    const prev = decoded;
    68→    decoded = decoded
    69→      .replace(/&amp;/g, '&')
    70→      .replace(/&lt;/g, '<')
    71→      .replace(/&gt;/g, '>')
    72→      .replace(/&quot;/g, '"')
    73→      .replace(/&#39;/g, "'")
    74→      .replace(/&nbsp;/g, ' ');
    75→    if (decoded === prev) break;
    76→  }
    77→  return decoded;
    78→}

    80→// Parse YouTube URL to get embed URL
    81→function getYouTubeEmbedUrl(url: string): string | null {
    82→  try {
    83→    const urlObj = new URL(url);
    84→    // Standard youtube.com/watch?v=...
    85→    if (urlObj.hostname.includes('youtube.com') && urlObj.searchParams.get('v')) {
    86→      return `https://www.youtube.com/embed/${urlObj.searchParams.get('v')}`;
    87→    }
    88→    // youtu.be/...
    89→    if (urlObj.hostname === 'youtu.be') {
    90→      return `https://www.youtube.com/embed/${urlObj.pathname.slice(1)}`;
    91→    }
    92→    // youtube.com/embed/...
    93→    if (urlObj.hostname.includes('youtube.com') && urlObj.pathname.startsWith('/embed/')) {
    94→      return url;
    95→    }
    96→    // Vimeo
    97→    if (urlObj.hostname.includes('vimeo.com')) {
    98→      const videoId = urlObj.pathname.split('/').filter(Boolean).pop();
    99→      return videoId ? `https://player.vimeo.com/video/${videoId}` : null;
   100→    }
   101→    return null;
   102→  } catch {
   103→    return null;
   104→  }
   105→}

   107→function sanitizeHtml(html: string): string {
   108→  if (!html) return '';
   109→  let sanitized = decodeHtmlEntities(html);
   110→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
   111→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
   112→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
   113→  return sanitized;
   114→}

   132→// Star Rating component
   133→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
   134→  const stars = [];
   135→  const fullStars = Math.floor(rating);
   136→  const hasHalf = rating - fullStars >= 0.5;
   137→  for (let i = 0; i < 5; i++) {
   138→    if (i < fullStars) stars.push('full');
   139→    else if (i === fullStars && hasHalf) stars.push('half');
   140→    else stars.push('empty');
   141→  }
   142→  return (
   143→    <div className="flex items-center gap-1">
   144→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
   145→        {stars.map((type, i) => (
   146→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-600'}`} />
   147→        ))}
   148→      </div>
   149→      {count !== undefined && (
   150→        <span className={`text-gray-500 dark:text-gray-300 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
   151→      )}
   152→    </div>
   153→  );
   154→}
