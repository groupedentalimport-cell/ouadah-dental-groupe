    48→// Animated counter component - counts up with smooth easing
    49→function AnimatedCounter({ end, suffix, delay = 0 }: { end: number; suffix: string; delay?: number }) {
    50→  const [count, setCount] = useState(end); // Show final value immediately
    51→  const [animated, setAnimated] = useState(false);
    52→  const hasStartedRef = useRef(false);
    53→
    54→  // Use IntersectionObserver directly for more reliable detection
    55→  const ref = useRef<HTMLDivElement>(null);
    56→
    57→  useEffect(() => {
    58→    const el = ref.current;
    59→    if (!el || hasStartedRef.current) return;
    60→
    61→    const observer = new IntersectionObserver(
    62→      (entries) => {
    63→        entries.forEach((entry) => {
    64→          if (entry.isIntersecting && !hasStartedRef.current) {
    65→            hasStartedRef.current = true;
    66→            // Start from 0 and count up
    67→            setCount(0);
    68→            setAnimated(true);
    69→            const startTime = performance.now() + delay;
    70→            const duration = 1800;
    71→            let rafId: number;
    72→
    73→            const animate = (currentTime: number) => {
    74→              const elapsed = currentTime - startTime;
    75→              if (elapsed < 0) {
    76→                rafId = requestAnimationFrame(animate);
    77→                return;
    78→              }
    79→              const progress = Math.min(elapsed / duration, 1);
    80→              const eased = 1 - Math.pow(1 - progress, 3);
    81→              setCount(Math.round(eased * end));
    82→              if (progress < 1) {
    83→                rafId = requestAnimationFrame(animate);
    84→              }
    85→            };
    86→
    87→            rafId = requestAnimationFrame(animate);
    88→            observer.disconnect();
    89→          }
    90→        });
    91→      },
    92→      { threshold: 0.1 }
    93→    );
    94→
    95→    observer.observe(el);
    96→    return () => observer.disconnect();
    97→  }, [end, delay]);
    98→
    99→  return (
   100→    <div ref={ref} className="text-4xl sm:text-5xl lg:text-6xl font-bold text-white mb-1">
   101→      {count}<span style={{ color: '#F0C05A' }}>{suffix}</span>
   102→    </div>
   103→  );
   104→}
