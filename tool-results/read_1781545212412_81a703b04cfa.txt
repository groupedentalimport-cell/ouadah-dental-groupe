    48→    48→// Animated counter component - counts up with smooth easing
    49→    49→function AnimatedCounter({ end, suffix, delay = 0 }: { end: number; suffix: string; delay?: number }) {
    50→    50→  const [count, setCount] = useState(end); // Show final value immediately
    51→    51→  const [animated, setAnimated] = useState(false);
    52→    52→  const hasStartedRef = useRef(false);
    53→    53→
    54→    54→  // Use IntersectionObserver directly for more reliable detection
    55→    55→  const ref = useRef<HTMLDivElement>(null);
    56→    56→
    57→    57→  useEffect(() => {
    58→    58→    const el = ref.current;
    59→    59→    if (!el || hasStartedRef.current) return;
    60→    60→
    61→    61→    const observer = new IntersectionObserver(
    62→    62→      (entries) => {
    63→    63→        entries.forEach((entry) => {
    64→    64→          if (entry.isIntersecting && !hasStartedRef.current) {
    65→    65→            hasStartedRef.current = true;
    66→    66→            // Start from 0 and count up
    67→    67→            setCount(0);
    68→    68→            setAnimated(true);
    69→    69→            const startTime = performance.now() + delay;
    70→    70→            const duration = 1800;
    71→    71→            let rafId: number;
    72→    72→
    73→    73→            const animate = (currentTime: number) => {
    74→    74→              const elapsed = currentTime - startTime;
    75→    75→              if (elapsed < 0) {
    76→    76→                rafId = requestAnimationFrame(animate);
    77→    77→                return;
    78→    78→              }
    79→    79→              const progress = Math.min(elapsed / duration, 1);
    80→    80→              const eased = 1 - Math.pow(1 - progress, 3);
    81→    81→              setCount(Math.round(eased * end));
    82→    82→              if (progress < 1) {
    83→    83→                rafId = requestAnimationFrame(animate);
    84→    84→              }
    85→    85→            };
    86→    86→
    87→    87→            rafId = requestAnimationFrame(animate);
    88→    88→            observer.disconnect();
    89→    89→          }
    90→    90→        });
    91→    91→      },
    92→    92→      { threshold: 0.1 }
    93→    93→    );
    94→    94→
    95→    95→    observer.observe(el);
    96→    96→    return () => observer.disconnect();
    97→    97→  }, [end, delay]);
    98→    98→
    99→    99→  return (
   100→   100→    <div ref={ref} className="text-4xl sm:text-5xl lg:text-6xl font-bold text-white mb-1">
   101→   101→      {count}<span style={{ color: '#F0C05A' }}>{suffix}</span>
   102→   102→    </div>
   103→   103→  );
   104→   104→}
