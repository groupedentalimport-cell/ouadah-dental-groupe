    42→    42→// Animated counter hook
    43→    43→function useCounter(end: number, duration = 2000) {
    44→    44→  const [count, setCount] = useState(0);
    45→    45→  const ref = useRef<HTMLDivElement>(null);
    46→    46→  const inView = useInView(ref, { once: true });
    47→    47→
    48→    48→  useEffect(() => {
    49→    49→    if (!inView) return;
    50→    50→    let start = 0;
    51→    51→    const step = end / (duration / 16);
    52→    52→    const timer = setInterval(() => {
    53→    53→      start += step;
    54→    54→      if (start >= end) {
    55→    55→        setCount(end);
    56→    56→        clearInterval(timer);
    57→    57→      } else {
    58→    58→        setCount(Math.floor(start));
    59→    59→      }
    60→    60→    }, 16);
    61→    61→    return () => clearInterval(timer);
    62→    62→  }, [inView, end, duration]);
    63→    63→
    64→    64→  return { count, ref };
    65→    65→}
