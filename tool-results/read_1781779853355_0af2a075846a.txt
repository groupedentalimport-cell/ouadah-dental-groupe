    53→// Star Rating component
    54→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
    55→  const stars = [];
    56→  const fullStars = Math.floor(rating);
    57→  const hasHalf = rating - fullStars >= 0.5;
    58→  for (let i = 0; i < 5; i++) {
    59→    if (i < fullStars) stars.push('full');
    60→    else if (i === fullStars && hasHalf) stars.push('half');
    61→    else stars.push('empty');
    62→  }
    63→  return (
    64→    <div className="flex items-center gap-1">
    65→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
    66→        {stars.map((type, i) => (
    67→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-500'}`} />
    68→        ))}
    69→      </div>
    70→      {count !== undefined && (
    71→        <span className={`text-gray-500 dark:text-gray-300 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
    72→      )}
    73→    </div>
    74→  );
    75→}
