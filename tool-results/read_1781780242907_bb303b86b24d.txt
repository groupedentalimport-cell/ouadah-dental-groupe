   172→// Animated Counter Component
   173→function AnimatedCounter({ target, duration = 1500 }: { target: number; duration?: number }) {
   174→  const [count, setCount] = useState(0);
   175→  const ref = useRef<HTMLSpanElement>(null);
   176→  const hasAnimated = useRef(false);
   177→
   178→  useEffect(() => {
   179→    if (hasAnimated.current) return;
   180→    hasAnimated.current = true;
   181→    let start = 0;
   182→    const increment = target / (duration / 16);
   183→    const timer = setInterval(() => {
   184→      start += increment;
   185→      if (start >= target) {
   186→        setCount(target);
   187→        clearInterval(timer);
   188→      } else {
   189→        setCount(Math.floor(start));
   190→      }
   191→    }, 16);
   192→    return () => clearInterval(timer);
   193→  }, [target, duration]);
   194→
   195→  return <span ref={ref}>{count.toLocaleString()}</span>;
   196→}
