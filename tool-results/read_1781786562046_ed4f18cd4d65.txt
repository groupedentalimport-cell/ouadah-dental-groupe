    52→// Star Rating component
    53→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
    54→  const stars = [];
    55→  const fullStars = Math.floor(rating);
    56→  const hasHalf = rating - fullStars >= 0.5;
    57→  for (let i = 0; i < 5; i++) {
    58→    if (i < fullStars) stars.push('full');
    59→    else if (i === fullStars && hasHalf) stars.push('half');
    60→    else stars.push('empty');
    61→  }
    62→  return (
    63→    <div className="flex items-center gap-1">
    64→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
    65→        {stars.map((type, i) => (
    66→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-500'}`} />
    67→        ))}
    68→      </div>
    69→      {count !== undefined && (
    70→        <span className={`text-gray-500 dark:text-gray-300 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
    71→      )}
    72→    </div>
    73→  );
    74→}
