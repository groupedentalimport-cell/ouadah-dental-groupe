    24→    24→const RECENT_SEARCHES_KEY = 'odg-recent-searches';
    25→    25→const MAX_RECENT = 5;
    26→    26→
    27→    27→function getRecentSearches(): string[] {
    28→    28→  if (typeof window === 'undefined') return [];
    29→    29→  try {
    30→    30→    const stored = localStorage.getItem(RECENT_SEARCHES_KEY);
    31→    31→    return stored ? JSON.parse(stored) : [];
    32→    32→  } catch {
    33→    33→    return [];
    34→    34→  }
    35→    35→}
    36→    36→
    37→    37→function addRecentSearch(query: string) {
    38→    38→  if (!query.trim()) return;
    39→    39→  try {
    40→    40→    const existing = getRecentSearches();
    41→    41→    const filtered = existing.filter((s) => s.toLowerCase() !== query.toLowerCase());
    42→    42→    const updated = [query, ...filtered].slice(0, MAX_RECENT);
    43→    43→    localStorage.setItem(RECENT_SEARCHES_KEY, JSON.stringify(updated));
    44→    44→  } catch {
    45→    45→    // ignore
    46→    46→  }
    47→    47→}
    48→    48→
    49→    49→function removeRecentSearch(query: string) {
    50→    50→  try {
    51→    51→    const existing = getRecentSearches();
    52→    52→    const updated = existing.filter((s) => s !== query);
    53→    53→    localStorage.setItem(RECENT_SEARCHES_KEY, JSON.stringify(updated));
    54→    54→  } catch {
    55→    55→    // ignore
    56→    56→  }
    57→    57→}
