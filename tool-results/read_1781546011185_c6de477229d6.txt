    48→  includeOnQuote: boolean;

    50→}) {
    51→  return (
    52→    <div className="mt-4 pt-4 border-t border-gray-100 dark:border-white/10">
    53→      <h4 className="font-semibold text-xs uppercase tracking-wider mb-3 dark:text-[#7AB0E0]" style={{ color: '#1E3A5F' }}>

    57→        <Slider
    58→          value={priceRange}

    60→          min={0}
    61→          max={priceMax || 10000000}
    62→          step={Math.max(1, Math.floor((priceMax || 10000000) / 100))}
    63→          className="mb-3"
    64→        />
    65→        <div className="flex items-center gap-2">
    66→          <div className="flex-1">

    71→              onChange={(e) => {
    72→                const val = parseInt(e.target.value) || 0;
    73→                setPriceRange([Math.min(val, priceRange[1]), priceRange[1]]);
    74→              }}
    75→              className="h-8 text-xs mt-0.5 dark:bg-white/5 dark:border-white/10 dark:text-white"
    76→              min={0}
    77→              max={priceRange[1]}
    78→            />

    81→          <div className="flex-1">

    83→            <Input
    84→              type="number"
    85→              value={priceRange[1]}
    86→              onChange={(e) => {
    87→                const val = parseInt(e.target.value) || priceMax;
    88→                setPriceRange([priceRange[0], Math.max(val, priceRange[0])]);
    89→              }}
    90→              className="h-8 text-xs mt-0.5 dark:bg-white/5 dark:border-white/10 dark:text-white"
    91→              min={priceRange[0]}
    92→              max={priceMax}
    93→            />
    94→          </div>
    95→        </div>
    96→        {/* Include "Sur devis" toggle */}
    97→        <label className="flex items-center gap-2 mt-3 cursor-pointer group">
    98→          <input
    99→            type="checkbox"
   100→            checked={includeOnQuote}
   101→            onChange={(e) => setIncludeOnQuote(e.target.checked)}

   140→      if (p.brand) {
   141→        const key = p.brand.toLowerCase().trim();

   143→        if (existing) {
   144→          // Prefer the properly-cased version

   147→            existing.display = p.brand.trim();
   148→          }
   149→          existing.count += 1;
   150→        } else {
   151→          brandMap.set(key, { display: p.brand.trim(), count: 1 });
   152→        }
   153→      }
   154→    });

   156→      .map(([key, data]) => ({ brand: data.display, key, count: data.count }))

   197→

   206→          p.ref.toLowerCase().includes(q)

   227→        break;

   229→        result.sort((a, b) => (b.price || 0) - (a.price || 0));
   230→        break;

   232→        result.sort((a, b) =>
   233→          language === 'ar'
   234→            ? a.name_ar.localeCompare(b.name_ar)
   235→            : a.name_fr.localeCompare(b.name_fr)
   236→        );
   237→        break;

   282→          />
   283→        </div>
