    33→// Animated counter hook
    34→function useCounter(end: number, duration = 2000) {
    35→  const [count, setCount] = useState(0);
    36→  const ref = useRef<HTMLDivElement>(null);
    37→  const inView = useInView(ref, { once: true });
    38→
    39→  useEffect(() => {
    40→    if (!inView) return;
    41→    let start = 0;
    42→    const step = end / (duration / 16);
    43→    const timer = setInterval(() => {
    44→      start += step;
    45→      if (start >= end) {
    46→        setCount(end);
    47→        clearInterval(timer);
    48→      } else {
    49→        setCount(Math.floor(start));
    50→      }
    51→    }, 16);
    52→    return () => clearInterval(timer);
    53→  }, [inView, end, duration]);
    54→
    55→  return { count, ref };
    56→}
