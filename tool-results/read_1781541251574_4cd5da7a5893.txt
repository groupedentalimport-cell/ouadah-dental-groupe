    42→// Animated counter hook
    43→function useCounter(end: number, duration = 2000) {
    44→  const [count, setCount] = useState(0);
    45→  const ref = useRef<HTMLDivElement>(null);
    46→  const inView = useInView(ref, { once: true });
    47→
    48→  useEffect(() => {
    49→    if (!inView) return;
    50→    let start = 0;
    51→    const step = end / (duration / 16);
    52→    const timer = setInterval(() => {
    53→      start += step;
    54→      if (start >= end) {
    55→        setCount(end);
    56→        clearInterval(timer);
    57→      } else {
    58→        setCount(Math.floor(start));
    59→      }
    60→    }, 16);
    61→    return () => clearInterval(timer);
    62→  }, [inView, end, duration]);
    63→
    64→  return { count, ref };
    65→}
