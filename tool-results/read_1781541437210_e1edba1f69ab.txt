    41→// Decode double-encoded HTML entities
    42→function decodeHtmlEntities(text: string): string {
    43→  if (!text) return '';
    44→  let decoded = text;
    45→  for (let i = 0; i < 3; i++) {
    46→    const prev = decoded;
    47→    decoded = decoded
    48→      .replace(/&amp;/g, '&')
    49→      .replace(/&lt;/g, '<')
    50→      .replace(/&gt;/g, '>')
    51→      .replace(/&quot;/g, '"')
    52→      .replace(/&#39;/g, "'")
    53→      .replace(/&nbsp;/g, ' ');
    54→    if (decoded === prev) break;
    55→  }
    56→  return decoded;
    57→}
    58→
    59→function sanitizeHtml(html: string): string {
    60→  if (!html) return '';
    61→  let sanitized = decodeHtmlEntities(html);
    62→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
    63→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
    64→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
    65→  return sanitized;
    66→}
