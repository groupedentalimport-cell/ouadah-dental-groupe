  1127→export function DataProvider({ children }: { children: React.ReactNode }) {

  1130→  const [blogPosts, setBlogPosts] = useState<BlogPost[]>(defaultBlogPosts);
  1131→  const [contactInfo] = useState<ContactInfo>(defaultContactInfo);
  1132→  const [loading, setLoading] = useState(true);
  1133→  const [error, setError] = useState<string | null>(null);
  1134→
  1135→  const fetchData = useCallback(async () => {
  1136→    if (!isSupabaseConfigured || !supabase) {
  1137→      // Use default data when Supabase is not configured
  1138→      setLoading(false);
  1139→      return;
  1140→    }
  1141→
  1142→    setLoading(true);
  1143→    setError(null);
  1144→
  1145→    try {
  1146→      // Fetch categories - map Supabase column names to our interface
  1147→      const { data: catsData, error: catsError } = await supabase
  1148→        .from('categories')
  1149→        .select('*')
  1150→        .order('ordre', { ascending: true });
  1151→
  1152→      if (!catsError && catsData && catsData.length > 0) {
  1153→        const mapped = catsData.map((c: Record<string, unknown>) => ({
  1154→          id: c.id as string,
  1155→          slug: c.slug as string,
  1156→          name_fr: (c.nom_fr || c.name_fr || '') as string,
  1157→          name_ar: (c.nom_ar || c.name_ar || '') as string,
  1158→          description_fr: (c.description_fr || '') as string,
  1159→          description_ar: (c.description_ar || '') as string,
  1160→          icon: (c.icone || c.icon || 'Package') as string,
  1161→          image: (c.image || `/categories/${c.slug}.jpg`) as string,
  1162→          sort_order: (c.ordre || c.sort_order || 0) as number,
  1163→        }));
  1164→        setCategories(mapped);
  1165→      }
  1166→
  1167→      // Fetch products - map Supabase column names to our interface
  1168→      const { data: prodsData, error: prodsError } = await supabase
  1169→        .from('products')
  1170→        .select('*')
  1171→        .order('created_at', { ascending: false });
  1172→
  1173→      if (!prodsError && prodsData && prodsData.length > 0) {
  1174→        const mapped = prodsData.map((p: Record<string, unknown>) => ({
  1175→          id: p.id as string,
  1176→          slug: p.slug as string,
  1177→          name_fr: (p.nom_fr || p.name_fr || '') as string,
  1178→          name_ar: (p.nom_ar || p.name_ar || '') as string,
  1179→          description_fr: (p.description_fr || '') as string,
  1180→          description_ar: (p.description_ar || '') as string,

  1182→          short_description_ar: (p.short_description_ar || (p.description_ar as string || '').substring(0, 120)) as string,
  1183→          category_id: p.category_id as string,
  1184→          price: 0 as number, // Price not stored in Supabase - set on admin
  1185→          images: ((p.images || []) as string[]).map((img: string) => {
  1186→            // If already a full URL, return as-is
  1187→            if (img.startsWith('http')) return img;
  1188→            // Otherwise prefix with Supabase storage URL
  1189→            const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
  1190→            return `${supabaseUrl}/storage/v1/object/public/product-images/${img}`;
  1191→          }),
  1192→          specs: (p.specs || []) as Record<string, string>[],
  1193→          features_fr: (p.features_fr || []) as string[],
  1194→          features_ar: (p.features_ar || []) as string[],
  1195→          in_stock: (p.disponible ?? p.in_stock ?? true) as boolean,
