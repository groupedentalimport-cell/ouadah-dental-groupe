    38→    38→// Animated counter hook
    39→    39→function useCounter(end: number, duration = 2000) {
    40→    40→  const [count, setCount] = useState(0);
    41→    41→  const ref = useRef<HTMLDivElement>(null);
    42→    42→  const inView = useInView(ref, { once: true });
    43→    43→
    44→    44→  useEffect(() => {
    45→    45→    if (!inView) return;
    46→    46→    let start = 0;
    47→    47→    const step = end / (duration / 16);
    48→    48→    const timer = setInterval(() => {
    49→    49→      start += step;
    50→    50→      if (start >= end) {
    51→    51→        setCount(end);
    52→    52→        clearInterval(timer);
    53→    53→      } else {
    54→    54→        setCount(Math.floor(start));
    55→    55→      }
    56→    56→    }, 16);
    57→    57→    return () => clearInterval(timer);
    58→    58→  }, [inView, end, duration]);
    59→    59→
    60→    60→  return { count, ref };
    61→    61→}
