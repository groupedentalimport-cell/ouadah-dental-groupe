    49→    49→// Animated counter component - counts up with smooth easing
    50→    50→function AnimatedCounter({ end, suffix, delay = 0 }: { end: number; suffix: string; delay?: number }) {
    51→    51→  const [count, setCount] = useState(end); // Show final value immediately
    52→    52→  const [animated, setAnimated] = useState(false);
    53→    53→  const hasStartedRef = useRef(false);
    54→    54→
    55→    55→  // Use IntersectionObserver directly for more reliable detection
    56→    56→  const ref = useRef<HTMLDivElement>(null);
    57→    57→
    58→    58→  useEffect(() => {
    59→    59→    const el = ref.current;
    60→    60→    if (!el || hasStartedRef.current) return;
    61→    61→
    62→    62→    const observer = new IntersectionObserver(
    63→    63→      (entries) => {
    64→    64→        entries.forEach((entry) => {
    65→    65→          if (entry.isIntersecting && !hasStartedRef.current) {
    66→    66→            hasStartedRef.current = true;
    67→    67→            // Start from 0 and count up
    68→    68→            setCount(0);
    69→    69→            setAnimated(true);
    70→    70→            const startTime = performance.now() + delay;
    71→    71→            const duration = 1800;
    72→    72→            let rafId: number;
    73→    73→
    74→    74→            const animate = (currentTime: number) => {
    75→    75→              const elapsed = currentTime - startTime;
    76→    76→              if (elapsed < 0) {
    77→    77→                rafId = requestAnimationFrame(animate);
    78→    78→                return;
    79→    79→              }
    80→    80→              const progress = Math.min(elapsed / duration, 1);
    81→    81→              const eased = 1 - Math.pow(1 - progress, 3);
    82→    82→              setCount(Math.round(eased * end));
    83→    83→              if (progress < 1) {
    84→    84→                rafId = requestAnimationFrame(animate);
    85→    85→              }
    86→    86→            };
    87→    87→
    88→    88→            rafId = requestAnimationFrame(animate);
    89→    89→            observer.disconnect();
    90→    90→          }
    91→    91→        });
    92→    92→      },
    93→    93→      { threshold: 0.1 }
    94→    94→    );
    95→    95→
    96→    96→    observer.observe(el);
    97→    97→    return () => observer.disconnect();
    98→    98→  }, [end, delay]);
    99→    99→
   100→   100→  return (
   101→   101→    <div ref={ref} className="text-4xl sm:text-5xl lg:text-6xl font-bold text-white mb-1">
   102→   102→      {count}<span style={{ color: '#F0C05A' }}>{suffix}</span>
   103→   103→    </div>
   104→   104→  );
   105→   105→}

   150→   150→// Star Rating component
   151→   151→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
   152→   152→  const stars = [];
   153→   153→  const fullStars = Math.floor(rating);
   154→   154→  const hasHalf = rating - fullStars >= 0.5;
   155→   155→  for (let i = 0; i < 5; i++) {
   156→   156→    if (i < fullStars) stars.push('full');
   157→   157→    else if (i === fullStars && hasHalf) stars.push('half');
   158→   158→    else stars.push('empty');
   159→   159→  }
   160→   160→  return (
   161→   161→    <div className="flex items-center gap-1">
   162→   162→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
   163→   163→        {stars.map((type, i) => (
   164→   164→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-600'}`} />
   165→   165→        ))}
   166→   166→      </div>
   167→   167→      {count !== undefined && (
   168→   168→        <span className={`text-gray-500 dark:text-gray-400 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
   169→   169→      )}
   170→   170→    </div>
   171→   171→  );
   172→   172→}
