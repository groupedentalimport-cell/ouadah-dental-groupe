    56→// Decode double-encoded HTML entities
    57→function decodeHtmlEntities(text: string): string {
    58→  if (!text) return '';
    59→  let decoded = text;
    60→  for (let i = 0; i < 3; i++) {
    61→    const prev = decoded;
    62→    decoded = decoded
    63→      .replace(/&amp;/g, '&')
    64→      .replace(/&lt;/g, '<')
    65→      .replace(/&gt;/g, '>')
    66→      .replace(/&quot;/g, '"')
    67→      .replace(/&#39;/g, "'")
    68→      .replace(/&nbsp;/g, ' ');
    69→    if (decoded === prev) break;
    70→  }
    71→  return decoded;
    72→}
    73→
    74→function sanitizeHtml(html: string): string {
    75→  if (!html) return '';
    76→  let sanitized = decodeHtmlEntities(html);
    77→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
    78→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
    79→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
    80→  return sanitized;
    81→}

    99→// Star Rating component
   100→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
   101→  const stars = [];
   102→  const fullStars = Math.floor(rating);
   103→  const hasHalf = rating - fullStars >= 0.5;
   104→  for (let i = 0; i < 5; i++) {
   105→    if (i < fullStars) stars.push('full');
   106→    else if (i === fullStars && hasHalf) stars.push('half');
   107→    else stars.push('empty');
   108→  }
   109→  return (
   110→    <div className="flex items-center gap-1">
   111→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
   112→        {stars.map((type, i) => (
   113→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-600'}`} />
   114→        ))}
   115→      </div>
   116→      {count !== undefined && (
   117→        <span className={`text-gray-500 dark:text-gray-400 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
   118→      )}
   119→    </div>
   120→  );
   121→}
