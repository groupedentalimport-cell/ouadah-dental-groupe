    49→// Animated counter component - counts up with smooth easing
    50→function AnimatedCounter({ end, suffix, delay = 0 }: { end: number; suffix: string; delay?: number }) {
    51→  const [count, setCount] = useState(end); // Show final value immediately
    52→  const [animated, setAnimated] = useState(false);
    53→  const hasStartedRef = useRef(false);
    54→
    55→  // Use IntersectionObserver directly for more reliable detection
    56→  const ref = useRef<HTMLDivElement>(null);
    57→
    58→  useEffect(() => {
    59→    const el = ref.current;
    60→    if (!el || hasStartedRef.current) return;
    61→
    62→    const observer = new IntersectionObserver(
    63→      (entries) => {
    64→        entries.forEach((entry) => {
    65→          if (entry.isIntersecting && !hasStartedRef.current) {
    66→            hasStartedRef.current = true;
    67→            // Start from 0 and count up
    68→            setCount(0);
    69→            setAnimated(true);
    70→            const startTime = performance.now() + delay;
    71→            const duration = 1800;
    72→            let rafId: number;
    73→
    74→            const animate = (currentTime: number) => {
    75→              const elapsed = currentTime - startTime;
    76→              if (elapsed < 0) {
    77→                rafId = requestAnimationFrame(animate);
    78→                return;
    79→              }
    80→              const progress = Math.min(elapsed / duration, 1);
    81→              const eased = 1 - Math.pow(1 - progress, 3);
    82→              setCount(Math.round(eased * end));
    83→              if (progress < 1) {
    84→                rafId = requestAnimationFrame(animate);
    85→              }
    86→            };
    87→
    88→            rafId = requestAnimationFrame(animate);
    89→            observer.disconnect();
    90→          }
    91→        });
    92→      },
    93→      { threshold: 0.1 }
    94→    );
    95→
    96→    observer.observe(el);
    97→    return () => observer.disconnect();
    98→  }, [end, delay]);
    99→
   100→  return (
   101→    <div ref={ref} className="text-4xl sm:text-5xl lg:text-6xl font-bold text-white mb-1">
   102→      {count}<span style={{ color: '#F0C05A' }}>{suffix}</span>
   103→    </div>
   104→  );
   105→}
