    45→// Animated counter hook - shows end value, then animates from 0 when in view
    46→function useCounter(end: number, duration = 2000) {
    47→  const [count, setCount] = useState(0);
    48→  const hasAnimatedRef = useRef(false);
    49→  const ref = useRef<HTMLDivElement>(null);
    50→  const inView = useInView(ref, { once: true });
    51→
    52→  useEffect(() => {
    53→    if (!inView) return;
    54→    if (hasAnimatedRef.current) return;
    55→    hasAnimatedRef.current = true;
    56→    let start = 0;
    57→    const step = end / (duration / 16);
    58→    const timer = setInterval(() => {
    59→      start += step;
    60→      if (start >= end) {
    61→        setCount(end);
    62→        clearInterval(timer);
    63→      } else {
    64→        setCount(Math.floor(start));
    65→      }
    66→    }, 16);
    67→    return () => clearInterval(timer);
    68→  }, [inView, end, duration]);
    69→
    70→  return { count, ref };
    71→}
