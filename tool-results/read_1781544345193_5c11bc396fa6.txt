    45→  priceRange: [number, number];

    47→  priceMax: number;
    48→  includeOnQuote: boolean;
    49→  setIncludeOnQuote: (val: boolean) => void;
    50→}) {
    51→  return (
    52→    <div className="mt-4 pt-4 border-t border-gray-100 dark:border-white/10">
    53→      <h4 className="font-semibold text-xs uppercase tracking-wider mb-3 dark:text-[#7AB0E0]" style={{ color: '#1E3A5F' }}>

    57→        <Slider
    58→          value={priceRange}
    59→          onValueChange={(val) => setPriceRange([val[0], val[1]])}

    62→          step={Math.max(1, Math.floor((priceMax || 10000000) / 100))}
    63→          className="mb-3"
    64→        />
    65→        <div className="flex items-center gap-2">

    73→                setPriceRange([Math.min(val, priceRange[1]), priceRange[1]]);
    74→              }}
    75→              className="h-8 text-xs mt-0.5 dark:bg-white/5 dark:border-white/10 dark:text-white"
    76→              min={0}
    77→              max={priceRange[1]}
    78→            />

    83→            <Input
    84→              type="number"
    85→              value={priceRange[1]}
    86→              onChange={(e) => {
    87→                const val = parseInt(e.target.value) || priceMax;
    88→                setPriceRange([priceRange[0], Math.max(val, priceRange[0])]);

    90→              className="h-8 text-xs mt-0.5 dark:bg-white/5 dark:border-white/10 dark:text-white"
    91→              min={priceRange[0]}
    92→              max={priceMax}
    93→            />

    95→        </div>
    96→        {/* Include "Sur devis" toggle */}
    97→        <label className="flex items-center gap-2 mt-3 cursor-pointer group">

    99→            type="checkbox"
   100→            checked={includeOnQuote}
   101→            onChange={(e) => setIncludeOnQuote(e.target.checked)}

   495→                  <Stethoscope className="w-12 h-12" style={{ color: '#1E3A5F', opacity: 0.4 }} />
   496→                </div>
   497→                <p className="text-gray-500 dark:text-gray-400 text-lg font-medium mb-2">{t('products.noResults')}</p>

   499→                  {t('products.noResultsTip')}
   500→                </p>
   501→                {isFiltering && (
   502→                  <Button
   503→                    variant="outline"
   504→                    onClick={handleClearFilters}
   505→                    className="gap-2 dark:border-white/10 dark:text-[#7AB0E0]"
   506→                    style={{ borderColor: '#1E3A5F', color: '#1E3A5F' }}
   507→                  >
   508→                    <RotateCcw className="w-4 h-4" />
   509→                    {t('products.clearFilters')}
   510→                  </Button>

   512→              </motion.div>
   513→            ) : (
   514→              <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-3 gap-4 sm:gap-5">
   515→                {filteredProducts.map((product, i) => (
   516→                  <ProductCard
   517→                    key={product.id}
   518→                    product={product}
   519→                    index={i}
   520→                    language={language}
   521→                    getProductName={getProductName}
   522→                    getProductShortDescription={getProductShortDescription}
   523→                    formatPrice={formatPrice}
   524→                    t={t}
   525→                    navigate={navigate}
   526→                    addItem={addItem}
   527→                    isInCompare={isInCompare}
   528→                    addCompareProduct={addCompareProduct}
   529→                    removeCompareProduct={removeCompareProduct}
   530→                    compareFull={productIds.length >= 4}
   531→                    onQuickView={() => {
   532→                      setQuickViewProduct(product);
   533→                      setQuickViewOpen(true);
   534→                      addRecentlyViewed(product.id, product.slug, getProductName(product, language), product.images?.[0] || '', product.price);
   535→                    }}
   536→                  />
   537→                ))}
   538→              </div>
   539→            )}
   540→          </div>
   541→        </div>
   542→      </div>
   543→      {/* Quick View Modal */}
   544→      <QuickViewModal
   545→        product={quickViewProduct}
   546→        open={quickViewOpen}
   547→        onClose={() => setQuickViewOpen(false)}
   548→      />
   549→    </div>
   550→    </ContentWithLoading>
   551→  );
