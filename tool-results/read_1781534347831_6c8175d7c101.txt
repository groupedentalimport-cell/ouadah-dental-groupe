   989→const translations: Record<Language, Translations> = { fr, ar };

   991→export function useTranslation() {
   992→  const context = useContext(I18nContext);
   993→  if (!context) {
   994→    throw new Error('useTranslation must be used within a LanguageProvider');
   995→  }
   996→  return context;
   997→}
   998→
   999→export function LanguageProvider({ children }: { children: React.ReactNode }) {
  1000→  // Always start with 'fr' to match the server render and avoid hydration mismatch.
  1001→  // The real language preference is read in useEffect after hydration.
  1002→  const [language, setLanguageState] = useState<Language>('fr');
  1003→  const [mounted, setMounted] = useState(false);
  1004→
  1005→  // After hydration, read the stored language preference
  1006→  useEffect(() => {
  1007→    let detectedLang: Language = 'fr';
  1008→
  1009→    // 1. Check user's explicit preference in localStorage
  1010→    const saved = localStorage.getItem('denta-lang') as Language | null;
  1011→    if (saved && (saved === 'fr' || saved === 'ar')) {
  1012→      detectedLang = saved;
  1013→    } else {
  1014→      // 2. Auto-detect from browser language
  1015→      const browserLangs = navigator.languages || [navigator.language];
  1016→      for (const lang of browserLangs) {
  1017→        const code = lang.toLowerCase().split('-')[0];
  1018→        if (code === 'ar') { detectedLang = 'ar'; break; }
  1019→        if (code === 'fr') { detectedLang = 'fr'; break; }
  1020→      }
  1021→
  1022→      // 3. Check primary browser language
  1023→      if (detectedLang === 'fr') {
  1024→        const primary = navigator.language?.toLowerCase() || '';
  1025→        if (primary.startsWith('ar')) detectedLang = 'ar';
  1026→      }
  1027→    }
  1028→
  1029→    // Only update if different from default to avoid unnecessary re-render
  1030→    if (detectedLang !== 'fr') {
  1031→      setLanguageState(detectedLang);
  1032→    }
  1033→    setMounted(true);
  1034→  }, []);
  1035→
  1036→  useEffect(() => {
  1037→    if (!mounted) return;
  1038→    document.documentElement.lang = language;
  1039→    document.documentElement.dir = language === 'ar' ? 'rtl' : 'ltr';
  1040→    if (language === 'ar') {
  1041→      document.documentElement.classList.add('rtl');
  1042→    } else {
  1043→      document.documentElement.classList.remove('rtl');
  1044→    }
  1045→  }, [language, mounted]);
  1046→
  1047→  const setLanguage = useCallback((lang: Language) => {
  1048→    setLanguageState(lang);
  1049→    localStorage.setItem('denta-lang', lang);
  1050→  }, []);
  1051→
  1052→  const t = useCallback(
  1053→    (key: string): string => {
  1054→      return translations[language][key] || key;
  1055→    },
  1056→    [language]
  1057→  );
  1058→
  1059→  const dir = language === 'ar' ? 'rtl' as const : 'ltr' as const;
  1060→  const isRTL = language === 'ar';
  1061→
  1062→  return (
  1063→    <I18nContext.Provider value={{ language, setLanguage, t, dir, isRTL }}>
  1064→      {children}
  1065→    </I18nContext.Provider>
  1066→  );
  1067→}
