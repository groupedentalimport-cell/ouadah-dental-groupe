     2→

     8→  GitCompareArrows,

    17→  MapPin,

    20→} from 'lucide-react';
    21→import { useCompare } from '@/hooks/useCompare';

    27→import { Card, CardContent } from '@/components/ui/card';

    29→
    30→// Safely get a spec value from a Record
    31→function getSpecField(spec: Record<string, string>, field: string): string {
    32→  return spec[field] || '';
    33→}
    34→
    35→// Star Rating Display Component
    36→function StarRating({ rating, maxStars = 5 }: { rating: number; maxStars?: number }) {
    37→  return (
    38→    <div className="flex items-center gap-0.5 justify-center">
    39→      {Array.from({ length: maxStars }, (_, i) => (
    40→        <Star
    41→          key={i}
    42→          className={`w-4 h-4 ${
    43→            i < Math.floor(rating)
    44→              ? 'text-amber-400 fill-amber-400'
    45→              : i < rating
    46→              ? 'text-amber-400 fill-amber-400/50'
    47→              : 'text-gray-300'
    48→          }`}
    49→        />
    50→      ))}
    51→      <span className="text-xs font-medium ml-1.5" style={{ color: '#1E3A5F' }}>
    52→        {rating.toFixed(1)}
    53→      </span>
    54→    </div>
    55→  );
    56→}
    57→
    58→export default function CompareView() {
    59→  const { t, language, isRTL } = useTranslation();
    60→  const { navigate } = useRouter();
    61→  const { productIds, removeProduct, clearCompare } = useCompare();
    62→  const { products, categories, getProductName, getCategoryName } = useData();
    63→
    64→  const compareProducts = useMemo(
    65→    () =>
    66→      productIds
    67→        .map((id) => products.find((p) => p.id === id))
    68→        .filter(Boolean) as Product[],
    69→    [productIds, products]
    70→  );
    71→
    72→  const formatPrice = (price: number | undefined) => {
    73→    if (price == null || isNaN(price) || price === 0) return null;
    74→    return price.toLocaleString('fr-DZ') + ' DA';
    75→  };
    76→
    77→  // Safely parse specs from a product
    78→  const safeSpecs = (product: Product): Record<string, string>[] => {
    79→    try {
    80→      const s = product.specs;
    81→      if (Array.isArray(s)) return s.filter((sp) => sp && typeof sp === 'object');
    82→      return [];
    83→    } catch {
    84→      return [];
    85→    }
    86→  };
    87→
    88→  // Collect all unique spec labels across compared products
    89→  const allSpecLabels = useMemo(() => {
    90→    const labelSet = new Map<string, { label_fr: string; label_ar: string }>();
    91→    try {
    92→      compareProducts.forEach((p) => {
    93→        safeSpecs(p).forEach((spec) => {
    94→          const key = getSpecField(spec, 'label_fr') || getSpecField(spec, 'label_ar') || getSpecField(spec, 'label') || '';
    95→          if (key && !labelSet.has(key)) {
    96→            labelSet.set(key, {
    97→              label_fr: getSpecField(spec, 'label_fr') || getSpecField(spec, 'label') || key,
    98→              label_ar: getSpecField(spec, 'label_ar') || getSpecField(spec, 'label') || key,
    99→            });
   100→          }
   101→        });
   102→      });
   103→    } catch {
   104→      // If spec parsing fails, return empty
   105→    }
   106→    return Array.from(labelSet.values());
   107→  }, [compareProducts]);
   108→
   109→  // Check if a spec value differs among products
   110→  const isSpecDifferent = (labelKey: string) => {
   111→    const values = compareProducts.map((p) => {
   112→      const spec = safeSpecs(p).find(
   113→        (s) => (getSpecField(s, 'label_fr') || getSpecField(s, 'label_ar') || getSpecField(s, 'label')) === labelKey
   114→      );
   115→      return spec ? getSpecField(spec, 'value') : '';
   116→    });
   117→    const uniqueValues = new Set(values);
   118→    return uniqueValues.size > 1;
   119→  };
   120→
   121→  // Collect all unique features across compared products
   122→  const allFeatures = useMemo(() => {
   123→    const featureSet = new Set<string>();
   124→    compareProducts.forEach((p) => {
   125→      const feats = language === 'ar' ? p.features_ar : p.features_fr;
   126→      (feats || []).forEach((f) => featureSet.add(f));
   127→    });
   128→    return Array.from(featureSet);
   129→  }, [compareProducts, language]);
   130→
   131→  // Check if a feature differs among products
   132→  const isFeatureDifferent = (feature: string) => {
   133→    const presences = compareProducts.map((p) => {
   134→      const feats = language === 'ar' ? p.features_ar : p.features_fr;
   135→      return (feats || []).includes(feature);
   136→    });
   137→    const unique = new Set(presences);
   138→    return unique.size > 1;
   139→  };
   140→
   141→  // Generic attribute difference checker
   142→  const isAttributeDifferent = (getter: (p: Product) => string) => {
   143→    const values = compareProducts.map(getter);
   144→    const unique = new Set(values);
   145→    return unique.size > 1;
   146→  };
   147→
   148→  if (compareProducts.length < 2) {
   149→    return (
   150→      <div className="min-h-screen bg-gray-50 pt-20 flex items-center justify-center">
   151→        <motion.div
   152→          initial={{ opacity: 0, y: 20 }}
   153→          animate={{ opacity: 1, y: 0 }}
   154→          className="text-center px-4"
   155→        >
   156→          <div
   157→            className="w-28 h-28 rounded-full mx-auto mb-6 flex items-center justify-center"
   158→            style={{ backgroundColor: '#f0f4f8' }}
   159→          >
   160→            <GitCompareArrows className="w-14 h-14" style={{ color: '#1E3A5F', opacity: 0.4 }} />
   161→          </div>
   162→          <h2 className="text-xl font-bold mb-2" style={{ color: '#1E3A5F' }}>
   163→            {t('compare.empty')}
   164→          </h2>
   165→          <p className="text-gray-500 mb-6">{t('compare.selectMin')}</p>
   166→          <Button
   167→            onClick={() => navigate('/catalogue')}
   168→            className="font-semibold shadow-md"
   169→            style={{ backgroundColor: '#1E3A5F', color: '#fff' }}
   170→          >
   171→            {t('compare.backToCatalogue')}
   172→          </Button>
   173→        </motion.div>
   174→      </div>
   175→    );
   176→  }
   177→
   178→  // Difference indicator component
   179→  const DiffIndicator = () => (
   180→    <span
   181→      className="inline-flex items-center justify-center w-4 h-4 rounded-full text-[9px] font-bold ml-1 shrink-0"
   182→      style={{ backgroundColor: 'rgba(240,192,90,0.2)', color: '#D4A017' }}
   183→    >
   184→      ≠
   185→    </span>
   186→  );
   187→
   188→  // Row definition helper
   189→  type RowDef = {
   190→    key: string;
   191→    label: string;
   192→    getValue: (p: Product) => React.ReactNode;
   193→    isDifferent?: boolean;
   194→  };
   195→
   196→  const rows: RowDef[] = [
   197→    {
   198→      key: 'image',
   199→      label: t('compare.image'),
   200→      getValue: (p) => (
   201→        <div className="w-full aspect-square max-w-[160px] mx-auto rounded-xl overflow-hidden bg-white shadow-sm border border-gray-100">
   202→          {p.images && p.images.length > 0 ? (
   203→            <img
   204→              src={p.images[0]}
   205→              alt={getProductName(p, language)}
   206→              loading="lazy"
   207→              className="w-full h-full object-cover"
   208→            />
   209→          ) : (
   210→            <div
   211→              className="w-full h-full flex items-center justify-center"
   212→              style={{
   213→                background: 'linear-gradient(135deg, #1E3A5F 0%, #2A5F8F 60%, #3A7CB8 100%)',
   214→              }}
   215→            >
   216→              <Package className="w-12 h-12 text-white/30" />
   217→            </div>
   218→          )}
   219→        </div>
   220→      ),
   221→    },
   222→    {
   223→      key: 'name',
   224→      label: t('compare.name'),
   225→      getValue: (p) => (
   226→        <span className="font-bold text-sm" style={{ color: '#1E3A5F' }}>
   227→          {getProductName(p, language)}
   228→        </span>
   229→      ),
   230→      isDifferent: isAttributeDifferent((p) => getProductName(p, language)),
   231→    },
   232→    {
   233→      key: 'rating',
   234→      label: t('compare.rating') || 'Note',
   235→      getValue: (p) => {
   236→        const rating = p.rating ?? (3 + Math.abs(hashCode(p.id)) % 20) / 10;
   237→        return <StarRating rating={rating} />;

   239→      isDifferent: isAttributeDifferent((p) => String(p.rating ?? 0)),
   240→    },
   241→    {
   242→      key: 'brand',
   243→      label: t('compare.brand'),
   244→      getValue: (p) =>
   245→        p.brand ? (
   246→          <div className="flex items-center justify-center gap-1.5">
   247→            <Tag className="w-3.5 h-3.5" style={{ color: '#F0C05A' }} />
   248→            <span className="text-sm font-medium">{p.brand}</span>
   249→          </div>
   250→        ) : (
   251→          <span className="text-gray-400 text-sm">—</span>
   252→        ),
   253→      isDifferent: isAttributeDifferent((p) => p.brand || ''),
   254→    },
   255→    {
   256→      key: 'price',
   257→      label: t('compare.price'),
   258→      getValue: (p) => {
   259→        const priceDisplay = formatPrice(p.price);
   260→        return priceDisplay ? (
   261→          <span className="font-bold text-lg" style={{ color: '#1E3A5F' }}>
   262→            {priceDisplay}
   263→          </span>
   264→        ) : (
   265→          <Badge
   266→            className="text-xs border-0 font-semibold"
   267→            style={{ backgroundColor: '#F0C05A', color: '#1E3A5F' }}
   268→          >
   269→            {t('products.priceOnQuote')}
   270→          </Badge>
   271→        );
   272→      },
   273→      isDifferent: isAttributeDifferent((p) => String(p.price || 0)),
   274→    },
   275→    {
   276→      key: 'category',
   277→      label: t('compare.category'),
   278→      getValue: (p) => {
   279→        const cat = categories.find((c) => c.id === p.category_id);
   280→        return cat ? (
   281→          <span className="text-sm">{getCategoryName(cat, language)}</span>
   282→        ) : (
   283→          <span className="text-gray-400 text-sm">—</span>
   284→        );
   285→      },
   286→      isDifferent: isAttributeDifferent((p) => p.category_id),
   287→    },
   288→    {
   289→      key: 'stock',
   290→      label: t('compare.stock'),
   291→      getValue: (p) =>
   292→        p.in_stock ? (
   293→          <div className="flex items-center justify-center gap-1.5">
   294→            <CircleDot className="w-3.5 h-3.5 text-green-600" />
   295→            <span className="text-sm font-medium text-green-700">{t('products.inStock')}</span>
   296→          </div>
   297→        ) : (
   298→          <div className="flex items-center justify-center gap-1.5">
   299→            <XCircle className="w-3.5 h-3.5 text-red-500" />
   300→            <span className="text-sm font-medium text-red-600">{t('products.outOfStock')}</span>
   301→          </div>
   302→        ),
   303→      isDifferent: isAttributeDifferent((p) => String(p.in_stock)),
   304→    },
   305→    {
   306→      key: 'warranty',
   307→      label: t('compare.warranty'),
   308→      getValue: (p) =>
   309→        p.warranty && p.warranty !== '-' ? (
   310→          <div className="flex items-center justify-center gap-1.5">
   311→            <Shield className="w-3.5 h-3.5 text-blue-600" />
   312→            <span className="text-sm font-medium">{p.warranty}</span>
   313→          </div>
   314→        ) : (
   315→          <span className="text-gray-400 text-sm">—</span>
   316→        ),
   317→      isDifferent: isAttributeDifferent((p) => p.warranty || ''),
   318→    },
   319→    {
   320→      key: 'origin',
   321→      label: t('compare.origin'),
   322→      getValue: (p) =>
   323→        p.origin ? (
   324→          <div className="flex items-center justify-center gap-1.5">
   325→            <MapPin className="w-3.5 h-3.5" style={{ color: '#1E3A5F' }} />
   326→            <span className="text-sm">{p.origin}</span>
   327→          </div>
   328→        ) : (
   329→          <span className="text-gray-400 text-sm">—</span>
   330→        ),
   331→      isDifferent: isAttributeDifferent((p) => p.origin || ''),
   332→    },
   333→    {
   334→      key: 'ref',
   335→      label: t('compare.ref'),
   336→      getValue: (p) =>
   337→        p.ref ? (
   338→          <div className="flex items-center justify-center gap-1.5">
   339→            <Hash className="w-3.5 h-3.5 text-gray-400" />
   340→            <span className="text-xs font-mono">{p.ref}</span>
   341→          </div>
   342→        ) : (
   343→          <span className="text-gray-400 text-sm">—</span>
   344→        ),
   345→      isDifferent: isAttributeDifferent((p) => p.ref || ''),
   346→    },
   347→  ];
   348→
   349→  return (
   350→    <div className="min-h-screen bg-gray-50 pt-20">
   351→      {/* Header */}
   352→      <div
   353→        className="relative overflow-hidden"
   354→        style={{
   355→          background: 'linear-gradient(135deg, #1E3A5F 0%, #2A5F8F 50%, #1E3A5F 100%)',
   356→        }}
   357→      >
   358→        <div className="absolute inset-0 overflow-hidden pointer-events-none">
   359→          <div
   360→            className="absolute -top-24 -right-24 w-64 h-64 rounded-full opacity-10"
   361→            style={{ backgroundColor: '#F0C05A' }}
   362→          />
   363→          <div
   364→            className="absolute -bottom-16 -left-16 w-48 h-48 rounded-full opacity-10"
   365→            style={{ backgroundColor: '#F0C05A' }}
   366→          />
   367→        </div>
   368→
   369→        <div className="relative max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8 sm:py-10">
   370→          <motion.div
   371→            initial={{ opacity: 0, y: 20 }}
   372→            animate={{ opacity: 1, y: 0 }}
   373→            className="text-center"
   374→          >
   375→            <div className="flex items-center justify-center gap-3 mb-3">
   376→              <GitCompareArrows className="w-8 h-8" style={{ color: '#F0C05A' }} />
   377→              <h1 className="text-3xl sm:text-4xl font-bold text-white">
   378→                {t('compare.title')}
   379→              </h1>
   380→            </div>
   381→            <p className="text-blue-200 text-sm">
   382→              {compareProducts.length} {t('compare.items')}
   383→            </p>
   384→            {/* Legend for difference highlighting */}
   385→            <div className="mt-3 flex items-center justify-center gap-4 text-xs text-blue-200/70">
   386→              <span className="flex items-center gap-1.5">
   387→                <span className="w-3 h-3 rounded-sm" style={{ backgroundColor: 'rgba(240,192,90,0.15)' }} />
   388→                {language === 'ar' ? 'قيم مختلفة' : 'Valeurs différentes'}
   389→              </span>
   390→              <span className="flex items-center gap-1">
   391→                <span
   392→                  className="inline-flex items-center justify-center w-3.5 h-3.5 rounded-full text-[7px] font-bold"
   393→                  style={{ backgroundColor: 'rgba(240,192,90,0.2)', color: '#D4A017' }}
   394→                >
   395→                  ≠
   396→                </span>
   397→                {language === 'ar' ? 'مختلف' : 'Diffère'}
   398→              </span>
   399→            </div>
   400→          </motion.div>
   401→        </div>
   402→      </div>
   403→
   404→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6">
   405→        {/* Back button + Clear */}
   406→        <div className="flex items-center justify-between mb-6">
   407→          <Button
   408→            variant="outline"
   409→            onClick={() => navigate('/catalogue')}
   410→            className="gap-2 shadow-sm"
   411→            style={{ borderColor: '#1E3A5F', color: '#1E3A5F' }}
   412→          >
   413→            <ArrowLeft className="w-4 h-4" />
   414→            {t('compare.backToCatalogue')}
   415→          </Button>
   416→          <Button
   417→            variant="ghost"
   418→            onClick={clearCompare}
   419→            className="gap-2 text-red-500 hover:text-red-700 hover:bg-red-50"
   420→          >
   421→            <GitCompareArrows className="w-4 h-4" />
   422→            {t('compare.clear')}
   423→          </Button>
   424→        </div>
   425→
   426→        {/* Comparison Table */}
   427→        <motion.div
   428→          initial={{ opacity: 0, y: 20 }}
   429→          animate={{ opacity: 1, y: 0 }}
   430→          transition={{ delay: 0.1 }}
   431→        >
   432→          <Card className="border-0 shadow-sm overflow-hidden">
   433→            <CardContent className="p-0">
   434→              <div className="overflow-x-auto relative">
   435→                {/* Gradient fade indicators for horizontal scroll on mobile */}
   436→                <div className="sticky left-0 top-0 bottom-0 w-6 z-20 pointer-events-none absolute bg-gradient-to-r from-white to-transparent sm:hidden" />
   437→                <table className="w-full border-collapse">
   438→                  <thead>
   439→                    <tr className="border-b" style={{ backgroundColor: '#f8fafc' }}>
   440→                      <th
   441→                        className="p-4 text-left text-sm font-semibold min-w-[140px] sticky left-0 z-10 border-r border-gray-100"
   442→                        style={{
   443→                          backgroundColor: '#f8fafc',
   444→                          color: '#1E3A5F',
   445→                        }}
   446→                      >
   447→                        <div className="flex items-center gap-2">
   448→                          <GitCompareArrows className="w-4 h-4" style={{ color: '#F0C05A' }} />
   449→                          {t('compare.highlightDiff')}
   450→                        </div>
   451→                      </th>
   452→                      {compareProducts.map((product) => (
   453→                        <th key={product.id} className="p-4 min-w-[200px]">
   454→                          <div className="flex flex-col items-center">
   455→                            {/* Product image in header */}
   456→                            <div className="w-16 h-16 rounded-xl overflow-hidden mb-2 shadow-sm border border-gray-100 bg-white">
   457→                              {product.images && product.images.length > 0 ? (
   458→                                <img
   459→                                  src={product.images[0]}
   460→                                  alt={getProductName(product, language)}
   461→                                  loading="lazy"
   462→                                  className="w-full h-full object-cover"
   463→                                />
   464→                              ) : (
   465→                                <div
   466→                                  className="w-full h-full flex items-center justify-center"
   467→                                  style={{ background: 'linear-gradient(135deg, #1E3A5F, #2A5F8F)' }}
   468→                                >
   469→                                  <Package className="w-5 h-5 text-white/40" />
   470→                                </div>
   471→                              )}
   472→                            </div>
   473→                            {/* Remove button */}
   474→                            <button
   475→                              onClick={() => removeProduct(product.id)}
   476→                              className="w-6 h-6 rounded-full bg-gray-100 hover:bg-red-100 text-gray-400 hover:text-red-500 flex items-center justify-center transition-colors"
   477→                            >
   478→                              <XCircle className="w-4 h-4" />
   479→                            </button>
   480→                          </div>
   481→                        </th>
   482→                      ))}
   483→                    </tr>
   484→                  </thead>
   485→                  <tbody>
   486→                    {rows.map((row) => (
   487→                      <tr
   488→                        key={row.key}
   489→                        className={`border-b transition-colors ${
   490→                          row.isDifferent
   491→                            ? ''
   492→                            : ''
   493→                        }`}
   494→                      >
   495→                        <td
   496→                          className={`p-4 text-sm font-medium sticky left-0 z-10 border-r border-gray-100 ${
   497→                            row.isDifferent ? '' : ''
   498→                          }`}
   499→                          style={{
   500→                            color: '#1E3A5F',
   501→                            backgroundColor: row.isDifferent ? '#FFFBEB' : '#f8fafc',
   502→                          }}
   503→                        >
   504→                          <div className="flex items-center gap-1.5">
   505→                            {row.label}
   506→                            {row.isDifferent && <DiffIndicator />}
   507→                          </div>
   508→                        </td>
   509→                        {compareProducts.map((product) => (
   510→                          <td
   511→                            key={product.id}
   512→                            className={`p-4 text-center ${
   513→                              row.isDifferent ? 'bg-amber-50/50' : ''
   514→                            }`}
   515→                          >
   516→                            {row.getValue(product)}
   517→                          </td>

   519→                      </tr>
   520→                    ))}
   521→
   522→                    {/* Specs section header */}
   523→                    {allSpecLabels.length > 0 && (
   524→                      <tr style={{ backgroundColor: '#1E3A5F' }}>
   525→                        <td
   526→                          className="p-3 text-sm font-bold text-white sticky left-0 z-10 border-r border-[#2A5F8F]"
   527→                          style={{ backgroundColor: '#1E3A5F' }}
   528→                        >
   529→                          {t('compare.specs')}
   530→                        </td>

   532→                          <td key={product.id} className="p-3" />
   533→                        ))}
   534→                      </tr>
   535→                    )}
   536→
   537→                    {/* Spec rows */}
   538→                    {allSpecLabels.map((specLabel) => {
   539→                      const labelKey = specLabel.label_fr || specLabel.label_ar;
   540→                      const different = isSpecDifferent(labelKey);
   541→                      return (
   542→                        <tr
   543→                          key={`spec-${labelKey}`}
   544→                          className={`border-b ${different ? '' : ''}`}

   546→                          <td
   547→                            className="p-3 text-sm sticky left-0 z-10 border-r border-gray-100"

   676→// Simple hash function for deterministic rating fallback
   677→function hashCode(str: string): number {
   678→  let hash = 0;
   679→  for (let i = 0; i < str.length; i++) {
   680→    const char = str.charCodeAt(i);
   681→    hash = ((hash << 5) - hash) + char;
   682→    hash |= 0;
   683→  }
   684→  return hash;
   685→}
