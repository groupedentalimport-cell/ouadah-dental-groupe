   725→const translations: Record<Language, Translations> = { fr, ar };
   726→
   727→function getStoredLanguage(): Language {
   728→  if (typeof window === 'undefined') return 'fr';
   729→  const stored = localStorage.getItem('odg_language') as Language;
   730→  if (stored && (stored === 'fr' || stored === 'ar')) return stored;
   731→  return 'fr';
   732→}
   733→
   734→export function LanguageProvider({ children }: { children: React.ReactNode }) {
   735→  const [language, setLanguage] = useState<Language>(getStoredLanguage);
   736→
   737→  useEffect(() => {
   738→    localStorage.setItem('odg_language', language);
   739→    document.documentElement.lang = language;
   740→    document.documentElement.dir = language === 'ar' ? 'rtl' : 'ltr';
   741→  }, [language]);
   742→
   743→  const t = useCallback(
   744→    (key: string): string => {
   745→      return translations[language]?.[key] || translations.fr?.[key] || key;
   746→    },
   747→    [language]
   748→  );
   749→
   750→  const isRTL = language === 'ar';
   751→  const dir = isRTL ? 'rtl' : 'ltr';
   752→
   753→  return (
   754→    <I18nContext.Provider value={{ language, setLanguage, t, dir, isRTL }}>
   755→      {children}
   756→    </I18nContext.Provider>
   757→  );
   758→}

   761→  return useContext(I18nContext);
