   456→
   457→function ProductCard({
   458→  product,
   459→  index,
   460→  language,
   461→  getProductName,
   462→  getProductShortDescription,
   463→  formatPrice,
   464→  t,
   465→  navigate,
   466→  addItem,
   467→  isInCompare,
   468→  addCompareProduct,
   469→  removeCompareProduct,
   470→  compareFull,
   471→  onQuickView,
   472→}: {
   473→  product: Product;
   474→  index: number;
   475→  language: string;
   476→  getProductName: (p: Product, lang: any) => string;
   477→  getProductShortDescription: (p: Product, lang: any) => string;
   478→  formatPrice: (p: number | undefined) => string | null;
   479→  t: (k: string) => string;
   480→  navigate: (p: string) => void;
   481→  addItem: (item: any) => void;
   482→  isInCompare: (id: string) => boolean;
   483→  addCompareProduct: (id: string) => void;
   484→  removeCompareProduct: (id: string) => void;
   485→  compareFull: boolean;
   486→  onQuickView: () => void;
   487→}) {
   488→  const priceDisplay = formatPrice(product.price);
   489→  const isOnQuote = !priceDisplay; // price is 0 or missing
   490→
   491→  return (
   492→    <motion.div
   493→      initial={{ opacity: 0, y: 20 }}
   494→      animate={{ opacity: 1, y: 0 }}
   495→      transition={{ delay: Math.min(index * 0.04, 0.3) }}
   496→    >
   497→      <Card
   498→        className="group border-0 shadow-sm hover:shadow-xl transition-all duration-300 overflow-hidden h-full flex flex-col cursor-pointer"
   499→        onClick={() => navigate(`/product/${product.slug}`)}
   500→      >
   501→        {/* Image area */}
   502→        <div className="h-48 relative overflow-hidden">
   503→          {product.images && product.images.length > 0 ? (
   504→            <img
   505→              src={product.images[0]}
   506→              alt={getProductName(product, language)}
   507→              className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
   508→            />
   509→          ) : (
   510→            <div
   511→              className="w-full h-full flex items-center justify-center"
   512→              style={{
   513→                background: 'linear-gradient(135deg, #1E3A5F 0%, #2A5F8F 60%, #3A7CB8 100%)',
   514→              }}
   515→            >
   516→              <div className="text-center">
   517→                <Package className="w-14 h-14 text-white/40 mx-auto mb-1" />
   518→                <p className="text-white/30 text-xs font-medium tracking-wider uppercase">ODG</p>
   519→              </div>
   520→            </div>
   521→          )}
   522→
   523→          {/* Top-left badges */}
   524→          <div className="absolute top-3 left-3 flex flex-col gap-1.5">
   525→            {product.is_new && (
   526→              <Badge className="text-[10px] px-2 py-0.5 border-0" style={{ backgroundColor: '#1E3A5F' }}>
   527→                {t('products.new')}
   528→              </Badge>
   529→            )}
   530→            {product.is_best_seller && (
   531→              <Badge className="text-[10px] px-2 py-0.5 border-0" style={{ backgroundColor: '#F0C05A', color: '#1E3A5F' }}>
   532→                {t('products.bestSeller')}
   533→              </Badge>
   534→            )}
   535→          </div>
   536→
   537→          {/* Top-right: Stock badge */}
   538→          <div className="absolute top-3 right-3">
