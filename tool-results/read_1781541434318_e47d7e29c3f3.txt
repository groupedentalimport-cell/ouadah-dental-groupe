   187→   187→function AdminLogin({ onLogin, t }: { onLogin: (p: string) => boolean; t: (k: string) => string }) {
   188→   188→  const [password, setPassword] = useState('');
   189→   189→  const [error, setError] = useState(false);
   190→   190→
   191→   191→  const handleSubmit = (e: React.FormEvent) => {
   192→   192→    e.preventDefault();
   193→   193→    if (!onLogin(password)) {
   194→   194→      setError(true);
   195→   195→      setPassword('');
   196→   196→    }
   197→   197→  };
   198→   198→
   199→   199→  return (
   200→   200→    <div className="min-h-screen flex items-center justify-center bg-gray-100 px-4">
   201→   201→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   202→   202→        <Card className="w-full max-w-sm border-0 shadow-xl">
   203→   203→          <CardContent className="p-8 text-center">
   204→   204→            <div
   205→   205→              className="w-16 h-16 rounded-2xl flex items-center justify-center mx-auto mb-6 text-white font-bold text-xl"
   206→   206→              style={{ backgroundColor: '#1E3A5F' }}
   207→   207→            >
   208→   208→              ODG
   209→   209→            </div>

   211→   211→              {t('admin.login.title')}
   212→   212→            </h1>
   213→   213→            <form onSubmit={handleSubmit} className="space-y-4">
   214→   214→              <div>
   215→   215→                <Input
   216→   216→                  type="password"
   217→   217→                  placeholder={t('admin.login.password')}
   218→   218→                  value={password}
   219→   219→                  onChange={(e) => {
   220→   220→                    setPassword(e.target.value);

   226→   226→              {error && (

   294→   294→      category_id: product.category_id,
   295→   295→      description_fr: product.description_fr,
