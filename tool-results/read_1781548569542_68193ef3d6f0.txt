    39→// Star Rating component
    40→function StarRating({ rating, count, size = 'sm' }: { rating: number; count?: number; size?: 'sm' | 'md' }) {
    41→  const stars = [];
    42→  const fullStars = Math.floor(rating);
    43→  const hasHalf = rating - fullStars >= 0.5;
    44→  for (let i = 0; i < 5; i++) {
    45→    if (i < fullStars) stars.push('full');
    46→    else if (i === fullStars && hasHalf) stars.push('half');
    47→    else stars.push('empty');
    48→  }
    49→  return (
    50→    <div className="flex items-center gap-1">
    51→      <div className={`flex ${size === 'sm' ? 'gap-0.5' : 'gap-1'}`}>
    52→        {stars.map((type, i) => (
    53→          <Star key={i} className={`${size === 'sm' ? 'w-3 h-3' : 'w-4 h-4'} ${type === 'full' ? 'text-[#F0C05A] fill-[#F0C05A]' : type === 'half' ? 'text-[#F0C05A]' : 'text-gray-300 dark:text-gray-600'}`} />
    54→        ))}
    55→      </div>
    56→      {count !== undefined && (
    57→        <span className={`text-gray-500 dark:text-gray-400 ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>({count})</span>
    58→      )}
    59→    </div>
    60→  );
    61→}
