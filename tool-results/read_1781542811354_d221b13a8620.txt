   333→// ============ Login Screen ============
   334→function AdminLogin({ onLogin, t }: { onLogin: (email: string, password: string) => boolean; t: (k: string) => string }) {
   335→  const [email, setEmail] = useState('');
   336→  const [password, setPassword] = useState('');
   337→  const [error, setError] = useState(false);
   338→  const [showError, setShowError] = useState(false);
   339→  const [showPassword, setShowPassword] = useState(false);
   340→
   341→  const handleSubmit = (e: React.FormEvent) => {
   342→    e.preventDefault();
   343→    if (!email.trim() || !password.trim()) {
   344→      setShowError(true);
   345→      return;
   346→    }
   347→    if (!onLogin(email, password)) {
   348→      setError(true);
   349→      setPassword('');
   350→      setShowError(true);
   351→    }
   352→  };
   353→
   354→  return (
   355→    <div
   356→      className="min-h-screen flex items-center justify-center px-4 relative overflow-hidden"
   357→      style={{ background: `linear-gradient(135deg, ${NAVY} 0%, #0F2440 50%, #1a3a5f 100%)` }}
   358→    >
   359→      {/* Dental-themed background pattern */}
   360→      <div className="absolute inset-0 opacity-[0.03]">
   361→        <svg width="100%" height="100%">
   362→          <defs>
   363→            <pattern id="dental-pattern" x="0" y="0" width="80" height="80" patternUnits="userSpaceOnUse">
   364→              <text x="10" y="30" fontSize="20" fill="white">🦷</text>
   365→              <text x="50" y="70" fontSize="16" fill="white">⚙</text>
   366→              <text x="35" y="45" fontSize="14" fill="white">✦</text>
   367→            </pattern>
   368→          </defs>
   369→          <rect width="100%" height="100%" fill="url(#dental-pattern)" />
   370→        </svg>
   371→      </div>
   372→
   373→      {/* Decorative floating elements */}
   374→      <div className="absolute top-20 left-20 w-32 h-32 rounded-full opacity-10" style={{ background: GOLD, filter: 'blur(60px)' }} />
   375→      <div className="absolute bottom-20 right-20 w-40 h-40 rounded-full opacity-10" style={{ background: '#3B8C6E', filter: 'blur(60px)' }} />
   376→
   377→      <motion.div
   378→        initial={{ opacity: 0, y: 30, scale: 0.95 }}
   379→        animate={{ opacity: 1, y: 0, scale: 1 }}
   380→        transition={{ duration: 0.5, ease: 'easeOut' }}
   381→        className="w-full max-w-md relative z-10"
   382→      >
   383→        <Card className="border-0 shadow-2xl overflow-hidden rounded-2xl">
   384→          {/* Gold accent line at top */}
   385→          <div className="h-1" style={{ background: `linear-gradient(90deg, ${GOLD}, #D4A84A, ${GOLD})` }} />
   386→          <CardContent className="p-8 sm:p-10">
   387→            {/* Logo */}
   388→            <div className="flex flex-col items-center mb-8">
   389→              <motion.div
   390→                initial={{ scale: 0.8 }}
   391→                animate={{ scale: 1 }}
   392→                transition={{ delay: 0.2, type: 'spring', stiffness: 200 }}
   393→                className="w-20 h-20 rounded-2xl flex items-center justify-center text-white font-bold text-2xl shadow-lg mb-4"
   394→                style={{ background: `linear-gradient(135deg, ${NAVY}, #2A5F8F)` }}
   395→              >
   396→                ODG
   397→              </motion.div>
   398→              <h1 className="text-2xl font-bold" style={{ color: NAVY }}>
   399→                {t('admin.login.title')}
   400→              </h1>
   401→              <p className="text-gray-400 text-sm mt-1">OUADAH Dental Groupe</p>
   402→            </div>
   403→
   404→            {/* Form */}
   405→            <form onSubmit={handleSubmit} className="space-y-4">
   406→              <div className="space-y-1.5">
   407→                <label className="text-sm font-medium" style={{ color: NAVY }}>
   408→                  Adresse e-mail
   409→                </label>
   410→                <div className="relative">
   411→                  <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   412→                  <Input
   413→                    type="email"
   414→                    placeholder="admin@odg.dz"
   415→                    value={email}
   416→                    onChange={(e) => {
   417→                      setEmail(e.target.value);
   418→                      setError(false);
   419→                      setShowError(false);
   420→                    }}
   421→                    className="pl-10 h-11 rounded-xl border-gray-200 focus:border-[#1E3A5F] focus:ring-[#1E3A5F]/20 transition-all"
   422→                    autoFocus
   423→                  />
   424→                </div>
   425→              </div>
   426→              <div className="space-y-1.5">
   427→                <label className="text-sm font-medium" style={{ color: NAVY }}>
   428→                  {t('admin.login.password')}
   429→                </label>
   430→                <div className="relative">
   431→                  <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   432→                  <Input
   433→                    type={showPassword ? 'text' : 'password'}
   434→                    placeholder="••••••••"
   435→                    value={password}
   436→                    onChange={(e) => {
   437→                      setPassword(e.target.value);
   438→                      setError(false);
   439→                      setShowError(false);
   440→                    }}
   441→                    className="pl-10 pr-10 h-11 rounded-xl border-gray-200 focus:border-[#1E3A5F] focus:ring-[#1E3A5F]/20 transition-all"
   442→                  />
   443→                  <button
   444→                    type="button"
   445→                    onClick={() => setShowPassword(!showPassword)}
   446→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   447→                  >
   448→                    {showPassword ? <Eye className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   449→                  </button>
   450→                </div>
   451→              </div>
   452→
   453→              <AnimatePresence>
   454→                {(error || showError) && (
   455→                  <motion.div
   456→                    initial={{ opacity: 0, height: 0 }}
   457→                    animate={{ opacity: 1, height: 'auto' }}
   458→                    exit={{ opacity: 0, height: 0 }}
   459→                  >
   460→                    <div className="flex items-center gap-2 p-3 rounded-lg text-sm" style={{ backgroundColor: '#FEF2F2', color: DANGER }}>
   461→                      <AlertCircle className="w-4 h-4 shrink-0" />
   462→                      {error ? 'Identifiants incorrects. Veuillez réessayer.' : 'Veuillez remplir tous les champs'}
   463→                    </div>
   464→                  </motion.div>
   465→                )}
   466→              </AnimatePresence>
   467→
   468→              <Button
   469→                type="submit"
   470→                className="w-full h-11 rounded-xl text-sm font-semibold shadow-lg hover:shadow-xl transition-all duration-300"
   471→                style={{ background: `linear-gradient(135deg, ${NAVY}, #2A5F8F)` }}
   472→              >
   473→                <Lock className="w-4 h-4 mr-2" />
   474→                {t('admin.login.submit')}
   475→              </Button>
   476→            </form>
   477→
   478→            {/* Forgot password link */}
   479→            <div className="mt-5 text-center">
   480→              <button className="text-xs text-gray-400 hover:text-[#1E3A5F] transition-colors hover:underline">
   481→                Mot de passe oublié ?
   482→              </button>
   483→            </div>
   484→          </CardContent>
   485→        </Card>
   486→      </motion.div>
   487→    </div>
   488→  );
   489→}

   711→      const matchesSearch = !searchQuery || name.includes(searchQuery.toLowerCase()) || (p.brand || '').toLowerCase().includes(searchQuery.toLowerCase());
   712→      const matchesCategory = filterCategory === 'all' || p.category_id === filterCategory;
   713→      return matchesSearch && matchesCategory;

   717→  const getCategoryName = (catId: string) => {
   718→    const cat = categories.find((c: any) => c.id === catId);
   719→    return cat ? cat.name_fr : '';
   720→  };
   721→
   722→  const startEdit = (product: any) => {
   723→    setEditProduct(product);

   725→      name_fr: product.name_fr,

   727→      price: String(product.price),
