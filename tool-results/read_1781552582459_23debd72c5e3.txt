    35→// Recent searches localStorage helpers
    36→const RECENT_SEARCHES_KEY = 'odg-recent-searches';
    37→const MAX_RECENT = 5;
    38→
    39→function getRecentSearches(): string[] {
    40→  if (typeof window === 'undefined') return [];
    41→  try {
    42→    const stored = localStorage.getItem(RECENT_SEARCHES_KEY);
    43→    return stored ? JSON.parse(stored) : [];
    44→  } catch {
    45→    return [];
    46→  }
    47→}
    48→
    49→function addRecentSearch(query: string) {
    50→  if (!query.trim()) return;
    51→  try {
    52→    const existing = getRecentSearches();
    53→    const filtered = existing.filter((s) => s.toLowerCase() !== query.toLowerCase());
    54→    const updated = [query, ...filtered].slice(0, MAX_RECENT);
    55→    localStorage.setItem(RECENT_SEARCHES_KEY, JSON.stringify(updated));
    56→  } catch {
    57→    // ignore
    58→  }
    59→}
    60→
    61→function removeRecentSearch(query: string) {
    62→  try {
    63→    const existing = getRecentSearches();
    64→    const updated = existing.filter((s) => s !== query);
    65→    localStorage.setItem(RECENT_SEARCHES_KEY, JSON.stringify(updated));
    66→  } catch {
    67→    // ignore
    68→  }
    69→}
