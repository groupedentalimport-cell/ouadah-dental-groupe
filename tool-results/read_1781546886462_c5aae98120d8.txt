    51→// Decode double-encoded HTML entities
    52→function decodeHtmlEntities(text: string): string {
    53→  if (!text) return '';
    54→  let decoded = text;
    55→  for (let i = 0; i < 3; i++) {
    56→    const prev = decoded;
    57→    decoded = decoded
    58→      .replace(/&amp;/g, '&')
    59→      .replace(/&lt;/g, '<')
    60→      .replace(/&gt;/g, '>')
    61→      .replace(/&quot;/g, '"')
    62→      .replace(/&#39;/g, "'")
    63→      .replace(/&nbsp;/g, ' ');
    64→    if (decoded === prev) break;
    65→  }
    66→  return decoded;
    67→}
    68→
    69→function sanitizeHtml(html: string): string {
    70→  if (!html) return '';
    71→  let sanitized = decodeHtmlEntities(html);
    72→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
    73→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
    74→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
    75→  return sanitized;
    76→}
