  1043→const translations: Record<Language, Translations> = { fr, ar };
  1044→
  1045→function getStoredLanguage(): Language {
  1046→  if (typeof window === 'undefined') return 'fr';
  1047→  const stored = localStorage.getItem('odg_language') as Language;
  1048→  if (stored && (stored === 'fr' || stored === 'ar')) return stored;
  1049→  return 'fr';
  1050→}
  1051→
  1052→export function LanguageProvider({ children }: { children: React.ReactNode }) {
  1053→  const [language, setLanguage] = useState<Language>(getStoredLanguage);
  1054→
  1055→  useEffect(() => {
  1056→    localStorage.setItem('odg_language', language);
  1057→    document.documentElement.lang = language;
  1058→    document.documentElement.dir = language === 'ar' ? 'rtl' : 'ltr';
  1059→  }, [language]);
  1060→
  1061→  const t = useCallback(
  1062→    (key: string): string => {
  1063→      return translations[language]?.[key] || translations.fr?.[key] || key;
  1064→    },
  1065→    [language]
  1066→  );
  1067→
  1068→  const isRTL = language === 'ar';
  1069→  const dir = isRTL ? 'rtl' : 'ltr';
  1070→
  1071→  return (
  1072→    <I18nContext.Provider value={{ language, setLanguage, t, dir, isRTL }}>
  1073→      {children}
  1074→    </I18nContext.Provider>
  1075→  );
  1076→}
  1077→
  1078→export function useTranslation() {
  1079→  return useContext(I18nContext);
