   440→
   441→function ProductCard({
   442→  product,
   443→  index,
   444→  language,
   445→  getProductName,
   446→  getProductShortDescription,
   447→  formatPrice,
   448→  t,
   449→  navigate,
   450→  addItem,
   451→  isInCompare,
   452→  addCompareProduct,
   453→  removeCompareProduct,
   454→  compareFull,
   455→}: {
   456→  product: Product;
   457→  index: number;
   458→  language: string;
   459→  getProductName: (p: Product, lang: any) => string;
   460→  getProductShortDescription: (p: Product, lang: any) => string;
   461→  formatPrice: (p: number | undefined) => string | null;
   462→  t: (k: string) => string;
   463→  navigate: (p: string) => void;
   464→  addItem: (item: any) => void;
   465→  isInCompare: (id: string) => boolean;
   466→  addCompareProduct: (id: string) => void;
   467→  removeCompareProduct: (id: string) => void;
   468→  compareFull: boolean;
   469→}) {
   470→  const priceDisplay = formatPrice(product.price);
   471→  const isOnQuote = !priceDisplay; // price is 0 or missing
   472→
   473→  return (
   474→    <motion.div
   475→      initial={{ opacity: 0, y: 20 }}
   476→      animate={{ opacity: 1, y: 0 }}
   477→      transition={{ delay: Math.min(index * 0.04, 0.3) }}
   478→    >
   479→      <Card
   480→        className="group border-0 shadow-sm hover:shadow-xl transition-all duration-300 overflow-hidden h-full flex flex-col cursor-pointer"
   481→        onClick={() => navigate(`/product/${product.slug}`)}
   482→      >
   483→        {/* Image area */}
   484→        <div className="h-48 relative overflow-hidden">
   485→          {product.images && product.images.length > 0 ? (
   486→            <img
   487→              src={product.images[0]}
   488→              alt={getProductName(product, language)}
   489→              className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
   490→            />
   491→          ) : (
   492→            <div
   493→              className="w-full h-full flex items-center justify-center"
   494→              style={{
   495→                background: 'linear-gradient(135deg, #1E3A5F 0%, #2A5F8F 60%, #3A7CB8 100%)',
   496→              }}
   497→            >
   498→              <div className="text-center">
   499→                <Package className="w-14 h-14 text-white/40 mx-auto mb-1" />
   500→                <p className="text-white/30 text-xs font-medium tracking-wider uppercase">ODG</p>
   501→              </div>
   502→            </div>
   503→          )}
   504→
   505→          {/* Top-left badges */}
   506→          <div className="absolute top-3 left-3 flex flex-col gap-1.5">
   507→            {product.is_new && (
   508→              <Badge className="text-[10px] px-2 py-0.5 border-0" style={{ backgroundColor: '#1E3A5F' }}>
   509→                {t('products.new')}
   510→              </Badge>
   511→            )}
   512→            {product.is_best_seller && (
   513→              <Badge className="text-[10px] px-2 py-0.5 border-0" style={{ backgroundColor: '#F0C05A', color: '#1E3A5F' }}>
   514→                {t('products.bestSeller')}
   515→              </Badge>
   516→            )}
   517→          </div>
   518→
   519→          {/* Top-right: Stock badge */}
   520→          <div className="absolute top-3 right-3">
   521→            {product.in_stock ? (
   522→              <Badge variant="outline" className="text-[10px] px-2 py-0.5 bg-white/90 border-green-200 text-green-700 backdrop-blur-sm">
   523→                {t('products.inStock')}
   524→              </Badge>
   525→            ) : (
   526→              <Badge variant="destructive" className="text-[10px] px-2 py-0.5">
   527→                {t('products.outOfStock')}
   528→              </Badge>
   529→            )}
   530→          </div>
   531→
   532→          {/* "Sur devis" badge when no price */}
   533→          {isOnQuote && (
   534→            <div className="absolute bottom-3 left-3">
   535→              <Badge
