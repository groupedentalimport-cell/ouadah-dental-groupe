    42→// Decode double-encoded HTML entities
    43→function decodeHtmlEntities(text: string): string {
    44→  if (!text) return '';
    45→  let decoded = text;
    46→  for (let i = 0; i < 3; i++) {
    47→    const prev = decoded;
    48→    decoded = decoded
    49→      .replace(/&amp;/g, '&')
    50→      .replace(/&lt;/g, '<')
    51→      .replace(/&gt;/g, '>')
    52→      .replace(/&quot;/g, '"')
    53→      .replace(/&#39;/g, "'")
    54→      .replace(/&nbsp;/g, ' ');
    55→    if (decoded === prev) break;
    56→  }
    57→  return decoded;
    58→}
    59→
    60→function sanitizeHtml(html: string): string {
    61→  if (!html) return '';
    62→  let sanitized = decodeHtmlEntities(html);
    63→  sanitized = sanitized.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
    64→  sanitized = sanitized.replace(/\s*on\w+\s*=\s*["'][^"']*["']/gi, '');
    65→  sanitized = sanitized.replace(/href\s*=\s*["']javascript:[^"']*["']/gi, '');
    66→  return sanitized;
    67→}
