   186→// ============ Login Screen ============
   187→function AdminLogin({ onLogin, t }: { onLogin: (p: string) => boolean; t: (k: string) => string }) {
   188→  const [password, setPassword] = useState('');
   189→  const [error, setError] = useState(false);
   190→
   191→  const handleSubmit = (e: React.FormEvent) => {
   192→    e.preventDefault();
   193→    if (!onLogin(password)) {
   194→      setError(true);
   195→      setPassword('');
   196→    }
   197→  };
   198→
   199→  return (
   200→    <div className="min-h-screen flex items-center justify-center bg-gray-100 px-4">
   201→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   202→        <Card className="w-full max-w-sm border-0 shadow-xl">
   203→          <CardContent className="p-8 text-center">
   204→            <div
   205→              className="w-16 h-16 rounded-2xl flex items-center justify-center mx-auto mb-6 text-white font-bold text-xl"
   206→              style={{ backgroundColor: '#1E3A5F' }}
   207→            >
   208→              ODG
   209→            </div>
   210→            <h1 className="text-xl font-bold mb-6" style={{ color: '#1E3A5F' }}>
   211→              {t('admin.login.title')}
   212→            </h1>
   213→            <form onSubmit={handleSubmit} className="space-y-4">
   214→              <div>
   215→                <Input
   216→                  type="password"
   217→                  placeholder={t('admin.login.password')}
   218→                  value={password}
   219→                  onChange={(e) => {
   220→                    setPassword(e.target.value);
   221→                    setError(false);
   222→                  }}
   223→                  className="text-center"
   224→                />
   225→              </div>
   226→              {error && (
   227→                <p className="text-red-500 text-sm">{t('admin.login.error')}</p>
   228→              )}
   229→              <Button type="submit" className="w-full" style={{ backgroundColor: '#1E3A5F' }}>
   230→                <Lock className="w-4 h-4 mr-2" />
   231→                {t('admin.login.submit')}
   232→              </Button>
   233→            </form>
   234→          </CardContent>
   235→        </Card>
   236→      </motion.div>
   237→    </div>
   238→  );
   239→}

   274→// ============ Products Tab ============
   275→function ProductsTab({ products, categories, getProductName, language, t, refresh }: any) {
   276→  const [showForm, setShowForm] = useState(false);
   277→  const [editProduct, setEditProduct] = useState<any>(null);
   278→  const [form, setForm] = useState({
   279→    name_fr: '',
   280→    name_ar: '',
   281→    price: '',
   282→    category_id: '',
   283→    description_fr: '',
   284→    brand: '',
   285→    ref: '',
   286→  });
   287→
   288→  const startEdit = (product: any) => {
   289→    setEditProduct(product);
   290→    setForm({
   291→      name_fr: product.name_fr,
   292→      name_ar: product.name_ar,
   293→      price: String(product.price),
   294→      category_id: product.category_id,
   295→      description_fr: product.description_fr,
   296→      brand: product.brand,
   297→      ref: product.ref,
   298→    });
   299→    setShowForm(true);
   300→  };
   301→
   302→  const resetForm = () => {
   303→    setForm({ name_fr: '', name_ar: '', price: '', category_id: '', description_fr: '', brand: '', ref: '' });
   304→    setEditProduct(null);
   305→    setShowForm(false);
   306→  };
   307→
   308→  const handleSave = async () => {
   309→    const { supabase, isSupabaseConfigured } = await import('@/lib/supabase');
   310→    if (!isSupabaseConfigured || !supabase) {
   311→      resetForm();
   312→      return;
   313→    }
   314→
   315→    const data = {
   316→      name_fr: form.name_fr,
   317→      name_ar: form.name_ar,
   318→      price: Number(form.price),
   319→      category_id: form.category_id,
   320→      description_fr: form.description_fr,
   321→      brand: form.brand,
   322→      ref: form.ref,
   323→    };
   324→
   325→    if (editProduct) {
   326→      await supabase.from('products').update(data).eq('id', editProduct.id);
   327→    } else {
   328→      await supabase.from('products').insert([{ ...data, slug: form.name_fr.toLowerCase().replace(/\s+/g, '-') }]);
   329→    }
   330→    refresh();
   331→    resetForm();
   332→  };
   333→
   334→  const handleDelete = async (id: string) => {
   335→    const { supabase, isSupabaseConfigured } = await import('@/lib/supabase');
   336→    if (!isSupabaseConfigured || !supabase) return;
   337→    await supabase.from('products').delete().eq('id', id);
   338→    refresh();
   339→  };
   340→
   341→  const formatPrice = (price: number | undefined) => {
   342→    if (price == null || isNaN(price)) return '— DA';
   343→    return price.toLocaleString('fr-DZ') + ' DA';
   344→  };
   345→
   346→  return (
   347→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-4">
   348→      <div className="flex items-center justify-between">
   349→        <h2 className="font-bold text-lg" style={{ color: '#1E3A5F' }}>
   350→          {t('admin.products.title')}
   351→        </h2>
   352→        <Button
   353→          onClick={() => { resetForm(); setShowForm(true); }}
   354→          className="gap-2"
   355→          style={{ backgroundColor: '#1E3A5F' }}
   356→        >
   357→          <Plus className="w-4 h-4" />
   358→          {t('admin.products.add')}
   359→        </Button>
   360→      </div>
   361→
   362→      {/* Form */}
   363→      <AnimatePresence>
   364→        {showForm && (
   365→          <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }}>
   366→            <Card className="border-0 shadow-sm">
   367→              <CardContent className="p-5 space-y-4">
   368→                <div className="flex items-center justify-between">
   369→                  <h3 className="font-semibold" style={{ color: '#1E3A5F' }}>
   370→                    {editProduct ? t('admin.products.edit') : t('admin.products.add')}
   371→                  </h3>
   372→                  <Button variant="ghost" size="icon" onClick={resetForm}>
   373→                    <X className="w-4 h-4" />
   374→                  </Button>
   375→                </div>
   376→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   377→                  <div>
   378→                    <label className="text-sm font-medium mb-1 block">{t('admin.products.name')} (FR)</label>
   379→                    <Input value={form.name_fr} onChange={(e) => setForm({ ...form, name_fr: e.target.value })} />
   380→                  </div>
   381→                  <div>
   382→                    <label className="text-sm font-medium mb-1 block">{t('admin.products.name')} (AR)</label>
   383→                    <Input value={form.name_ar} onChange={(e) => setForm({ ...form, name_ar: e.target.value })} dir="rtl" />
   384→                  </div>
   385→                  <div>
   386→                    <label className="text-sm font-medium mb-1 block">{t('admin.products.price')}</label>
   387→                    <Input type="number" value={form.price} onChange={(e) => setForm({ ...form, price: e.target.value })} />
   388→                  </div>
   389→                  <div>
   390→                    <label className="text-sm font-medium mb-1 block">{t('admin.products.category')}</label>
   391→                    <Select value={form.category_id} onValueChange={(v) => setForm({ ...form, category_id: v })}>
   392→                      <SelectTrigger><SelectValue /></SelectTrigger>
   393→                      <SelectContent>
   394→                        {categories.map((cat: any) => (
   395→                          <SelectItem key={cat.id} value={cat.id}>{cat.name_fr}</SelectItem>
   396→                        ))}
   397→                      </SelectContent>
   398→                    </Select>
   399→                  </div>
   400→                  <div>
   401→                    <label className="text-sm font-medium mb-1 block">{t('products.brand')}</label>
   402→                    <Input value={form.brand} onChange={(e) => setForm({ ...form, brand: e.target.value })} />
   403→                  </div>
   404→                  <div>
   405→                    <label className="text-sm font-medium mb-1 block">{t('products.ref')}</label>
   406→                    <Input value={form.ref} onChange={(e) => setForm({ ...form, ref: e.target.value })} />
   407→                  </div>
   408→                </div>
   409→                <div>
   410→                  <label className="text-sm font-medium mb-1 block">{t('admin.products.description')}</label>
   411→                  <Textarea rows={3} value={form.description_fr} onChange={(e) => setForm({ ...form, description_fr: e.target.value })} />
   412→                </div>
   413→                <div className="flex gap-2">
   414→                  <Button onClick={handleSave} className="gap-2" style={{ backgroundColor: '#1E3A5F' }}>
   415→                    <Save className="w-4 h-4" />
   416→                    {t('admin.products.save')}
   417→                  </Button>
   418→                  <Button variant="outline" onClick={resetForm}>{t('admin.products.cancel')}</Button>
   419→                </div>
   420→              </CardContent>
   421→            </Card>
   422→          </motion.div>
   423→        )}
   424→      </AnimatePresence>
   425→
   426→      {/* Product List */}
   427→      <div className="space-y-2">
   428→        {products.map((product: any) => (
   429→          <Card key={product.id} className="border-0 shadow-sm">
   430→            <CardContent className="p-4 flex items-center gap-4">
   431→              <div className="w-12 h-12 rounded-lg flex items-center justify-center shrink-0" style={{ backgroundColor: '#f0f4f8' }}>
   432→                <Package className="w-5 h-5 text-gray-400" />
   433→              </div>
   434→              <div className="flex-1 min-w-0">
   435→                <p className="font-semibold text-sm truncate" style={{ color: '#1E3A5F' }}>
   436→                  {getProductName(product, language)}
   437→                </p>
   438→                <p className="text-xs text-gray-500">{product.brand} • {product.ref}</p>
   439→              </div>
   440→              <span className="text-sm font-medium shrink-0">{formatPrice(product.price)}</span>
   441→              <div className="flex gap-1 shrink-0">
   442→                <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => startEdit(product)}>
   443→                  <Pencil className="w-3.5 h-3.5" />
   444→                </Button>
   445→                <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500" onClick={() => handleDelete(product.id)}>
   446→                  <Trash2 className="w-3.5 h-3.5" />
   447→                </Button>
   448→              </div>
   449→            </CardContent>
   450→          </Card>
   451→        ))}
   452→      </div>
   453→    </motion.div>
   454→  );
   455→}

   457→// ============ Blog Tab ============
   458→function BlogTab({ blogPosts, language, t, refresh }: any) {
   459→  const [showForm, setShowForm] = useState(false);
   460→  const [form, setForm] = useState({
   461→    title_fr: '',
   462→    title_ar: '',
   463→    excerpt_fr: '',
   464→    excerpt_ar: '',
   465→    content_fr: '',
   466→    content_ar: '',
   467→    category: '',
   468→    author: 'ODG Team',
   469→  });
   470→
   471→  const handleSave = async () => {
   472→    const { supabase, isSupabaseConfigured } = await import('@/lib/supabase');
   473→    if (!isSupabaseConfigured || !supabase) {
   474→      setShowForm(false);
   475→      return;
   476→    }
   477→
   478→    await supabase.from('blog_posts').insert([{
   479→      title_fr: form.title_fr,
   480→      title_ar: form.title_ar,
   481→      excerpt_fr: form.excerpt_fr,
   482→      excerpt_ar: form.excerpt_ar,
   483→      content_fr: form.content_fr,
   484→      content_ar: form.content_ar,
   485→      category: form.category,
   486→      author: form.author,
   487→      slug: form.title_fr.toLowerCase().replace(/\s+/g, '-').substring(0, 60),
   488→      date: new Date().toISOString().split('T')[0],
   489→      read_time: 5,
   490→      tags: [],
   491→    }]);
   492→    refresh();
   493→    setShowForm(false);
   494→    setForm({ title_fr: '', title_ar: '', excerpt_fr: '', excerpt_ar: '', content_fr: '', content_ar: '', category: '', author: 'ODG Team' });
   495→  };
   496→
   497→  const handleDelete = async (id: string) => {
   498→    const { supabase, isSupabaseConfigured } = await import('@/lib/supabase');
   499→    if (!isSupabaseConfigured || !supabase) return;
   500→    await supabase.from('blog_posts').delete().eq('id', id);
   501→    refresh();
   502→  };
   503→
   504→  const formatDate = (d: string) => {
   505→    try { return new Date(d).toLocaleDateString('fr-FR'); } catch { return d; }
   506→  };
   507→
   508→  return (
   509→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-4">
   510→      <div className="flex items-center justify-between">
   511→        <h2 className="font-bold text-lg" style={{ color: '#1E3A5F' }}>{t('admin.blog.title')}</h2>
   512→        <Button onClick={() => setShowForm(!showForm)} className="gap-2" style={{ backgroundColor: '#1E3A5F' }}>
   513→          <Plus className="w-4 h-4" />
   514→          {t('admin.blog.add')}

   585→function QuotesTab({ t, refresh }: { t: (k: string) => string; refresh: () => void }) {
   586→  const [quotes, setQuotes] = useState<any[]>([]);
   587→  const [loading, setLoading] = useState(true);
   588→
   589→  useEffect(() => {
   590→    const fetchQuotes = async () => {
   591→      const { supabase, isSupabaseConfigured } = await import('@/lib/supabase');
   592→      if (!isSupabaseConfigured || !supabase) {
   593→        setLoading(false);
   594→        return;
   595→      }
   596→      const { data } = await supabase.from('quotes').select('*').order('created_at', { ascending: false });
   597→      if (data) setQuotes(data);
   598→      setLoading(false);
   599→    };
   600→    fetchQuotes();
   601→  }, [refresh]);
   602→
   603→  const updateStatus = async (id: string, status: string) => {
   604→    const { supabase, isSupabaseConfigured } = await import('@/lib/supabase');
   605→    if (!isSupabaseConfigured || !supabase) return;
   606→    await supabase.from('quotes').update({ status }).eq('id', id);
   607→    setQuotes((prev) => prev.map((q) => (q.id === id ? { ...q, status } : q)));
   608→  };
   609→
   610→  const statusColors: Record<string, string> = {
   611→    new: 'bg-blue-100 text-blue-700',
   612→    in_progress: 'bg-yellow-100 text-yellow-700',
   613→    processed: 'bg-green-100 text-green-700',
   614→    archived: 'bg-gray-100 text-gray-700',
   615→  };
   616→
   617→  const statusLabels: Record<string, string> = {
   618→    new: t('admin.quotes.new'),
   619→    in_progress: t('admin.quotes.inProgress'),
   620→    processed: t('admin.quotes.processed'),
   621→    archived: t('admin.quotes.archived'),
   622→  };
   623→
   624→  if (loading) {
   625→    return <div className="text-center py-12 text-gray-400">{t('common.loading')}</div>;
   626→  }
   627→
   628→  return (
   629→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-4">
   630→      <h2 className="font-bold text-lg" style={{ color: '#1E3A5F' }}>{t('admin.quotes.title')}</h2>
   631→
   632→      {quotes.length === 0 ? (
   633→        <Card className="border-0 shadow-sm">
   634→          <CardContent className="p-12 text-center text-gray-400">
   635→            <ClipboardList className="w-12 h-12 mx-auto mb-3 opacity-50" />
   636→            <p>Aucun devis pour le moment</p>
   637→          </CardContent>
   638→        </Card>
   639→      ) : (
   640→        <div className="space-y-2">
   641→          {quotes.map((quote) => (
   642→            <Card key={quote.id} className="border-0 shadow-sm">
   643→              <CardContent className="p-4">
   644→                <div className="flex items-start justify-between gap-4">
   645→                  <div className="flex-1 min-w-0">
   646→                    <p className="font-semibold text-sm" style={{ color: '#1E3A5F' }}>{quote.nom}</p>
   647→                    <p className="text-xs text-gray-500">{quote.email} • {quote.telephone}</p>
   648→                    <p className="text-xs text-gray-400">{quote.wilaya} • {quote.type_client}</p>
   649→                    {quote.message && (
   650→                      <p className="text-xs text-gray-500 mt-1 line-clamp-2">{quote.message}</p>
   651→                    )}
   652→                  </div>
   653→                  <div className="flex flex-col items-end gap-2">
   654→                    <Badge className={statusColors[quote.status] || statusColors.new}>
   655→                      {statusLabels[quote.status] || quote.status}
   656→                    </Badge>
   657→                    <Select
   658→                      value={quote.status}
   659→                      onValueChange={(v) => updateStatus(quote.id, v)}
   660→                    >
   661→                      <SelectTrigger className="w-32 h-7 text-xs">
   662→                        <SelectValue />
   663→                      </SelectTrigger>
   664→                      <SelectContent>
   665→                        <SelectItem value="new">{t('admin.quotes.new')}</SelectItem>
   666→                        <SelectItem value="in_progress">{t('admin.quotes.inProgress')}</SelectItem>
   667→                        <SelectItem value="processed">{t('admin.quotes.processed')}</SelectItem>
   668→                        <SelectItem value="archived">{t('admin.quotes.archived')}</SelectItem>
   669→                      </SelectContent>
   670→                    </Select>
   671→                  </div>
   672→                </div>
   673→              </CardContent>
   674→            </Card>
   675→          ))}
   676→        </div>
   677→      )}
   678→    </motion.div>
   679→  );
