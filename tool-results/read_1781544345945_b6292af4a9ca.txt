    47→// Animated counter component - uses requestAnimationFrame for smooth counting
    48→function AnimatedCounter({ end, suffix, duration = 1800 }: { end: number; suffix: string; duration?: number }) {
    49→  const [count, setCount] = useState(0);
    50→  const hasAnimatedRef = useRef(false);
    51→  const ref = useRef<HTMLDivElement>(null);
    52→  const inView = useInView(ref, { once: true, margin: '-50px' });
    53→
    54→  useEffect(() => {
    55→    if (!inView || hasAnimatedRef.current) return;
    56→    hasAnimatedRef.current = true;
    57→    const startTime = performance.now();
    58→    let rafId: number;
    59→
    60→    const animate = (currentTime: number) => {
    61→      const elapsed = currentTime - startTime;
    62→      const progress = Math.min(elapsed / duration, 1);
    63→      // Ease out cubic
    64→      const eased = 1 - Math.pow(1 - progress, 3);
    65→      setCount(Math.round(eased * end));
    66→      if (progress < 1) {
    67→        rafId = requestAnimationFrame(animate);
    68→      }
    69→    };
    70→
    71→    rafId = requestAnimationFrame(animate);
    72→    return () => cancelAnimationFrame(rafId);
    73→  }, [inView, end, duration]);
    74→
    75→  return (
    76→    <div ref={ref} className="text-4xl sm:text-5xl lg:text-6xl font-bold text-white mb-1">
    77→      {count}<span style={{ color: '#F0C05A' }}>{suffix}</span>
    78→    </div>
    79→  );
    80→}
