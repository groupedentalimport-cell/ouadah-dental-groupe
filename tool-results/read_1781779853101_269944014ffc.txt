    30→
    31→// Safely get a spec value from a Record
    32→function getSpecField(spec: Record<string, string>, field: string): string {
    33→  return spec[field] || '';
    34→}
    35→
    36→// Star Rating Display Component
    37→function StarRating({ rating, maxStars = 5 }: { rating: number; maxStars?: number }) {
    38→  return (
    39→    <div className="flex items-center gap-0.5 justify-center">
    40→      {Array.from({ length: maxStars }, (_, i) => (
    41→        <Star
    42→          key={i}
    43→          className={`w-4 h-4 ${
    44→            i < Math.floor(rating)
    45→              ? 'text-amber-400 fill-amber-400'
    46→              : i < rating
    47→              ? 'text-amber-400 fill-amber-400/50'
    48→              : 'text-gray-300'
    49→          }`}
    50→        />
    51→      ))}
    52→      <span className="text-xs font-medium ml-1.5" style={{ color: '#1E3A5F' }}>
    53→        {rating.toFixed(1)}
    54→      </span>
    55→    </div>
    56→  );
    57→}
    58→
    59→export default function CompareView() {

    62→  const { productIds, removeProduct, clearCompare } = useCompare();
    63→  const { products, categories, getProductName, getCategoryName } = useData();
    64→  const addItem = useQuoteCart((s) => s.addItem);
    65→
    66→  const compareProducts = useMemo(
    67→    () =>
    68→      productIds
    69→        .map((id) => products.find((p) => p.id === id))
    70→        .filter(Boolean) as Product[],
    71→    [productIds, products]
    72→  );
    73→
    74→  const formatPrice = (price: number | undefined) => {
    75→    if (price == null || isNaN(price) || price === 0) return null;
    76→    return price.toLocaleString('fr-DZ') + ' DA';
    77→  };
    78→
    79→  // Safely parse specs from a product
    80→  const safeSpecs = (product: Product): Record<string, string>[] => {
    81→    try {
    82→      const s = product.specs;
    83→      if (Array.isArray(s)) return s.filter((sp) => sp && typeof sp === 'object');
    84→      return [];
    85→    } catch {
    86→      return [];
    87→    }
    88→  };
    89→
    90→  // Collect all unique spec labels across compared products
    91→  const allSpecLabels = useMemo(() => {
    92→    const labelSet = new Map<string, { label_fr: string; label_ar: string }>();
    93→    try {
    94→      compareProducts.forEach((p) => {
    95→        safeSpecs(p).forEach((spec) => {
    96→          const key = getSpecField(spec, 'label_fr') || getSpecField(spec, 'label_ar') || getSpecField(spec, 'label') || '';
    97→          if (key && !labelSet.has(key)) {
    98→            labelSet.set(key, {
    99→              label_fr: getSpecField(spec, 'label_fr') || getSpecField(spec, 'label') || key,
   100→              label_ar: getSpecField(spec, 'label_ar') || getSpecField(spec, 'label') || key,
   101→            });
   102→          }
   103→        });
   104→      });
   105→    } catch {
   106→      // If spec parsing fails, return empty
   107→    }
   108→    return Array.from(labelSet.values());
   109→  }, [compareProducts]);
   110→
   111→  // Check if a spec value differs among products
   112→  const isSpecDifferent = (labelKey: string) => {
   113→    const values = compareProducts.map((p) => {
   114→      const spec = safeSpecs(p).find(
   115→        (s) => (getSpecField(s, 'label_fr') || getSpecField(s, 'label_ar') || getSpecField(s, 'label')) === labelKey
   116→      );
   117→      return spec ? getSpecField(spec, 'value') : '';
   118→    });

   120→    return uniqueValues.size > 1;
   121→  };
   122→
   123→  // Collect all unique features across compared products
   124→  const allFeatures = useMemo(() => {
   125→    const featureSet = new Set<string>();
   126→    compareProducts.forEach((p) => {
   127→      const feats = language === 'ar' ? p.features_ar : p.features_fr;
   128→      (feats || []).forEach((f) => featureSet.add(f));
   129→    });
   130→    return Array.from(featureSet);
   131→  }, [compareProducts, language]);
   132→
   133→  // Check if a feature differs among products
   134→  const isFeatureDifferent = (feature: string) => {
   135→    const presences = compareProducts.map((p) => {
   136→      const feats = language === 'ar' ? p.features_ar : p.features_fr;
   137→      return (feats || []).includes(feature);
   138→    });
   139→    const unique = new Set(presences);
   140→    return unique.size > 1;
   141→  };
   142→
   143→  // Generic attribute difference checker
   144→  const isAttributeDifferent = (getter: (p: Product) => string) => {
   145→    const values = compareProducts.map(getter);
   146→    const unique = new Set(values);
   147→    return unique.size > 1;
   148→  };
   149→
   150→  if (compareProducts.length < 2) {
   151→    return (
   152→      <div className="min-h-screen bg-gray-50 pt-20 flex items-center justify-center">
   153→        <motion.div
   154→          initial={{ opacity: 0, y: 20 }}
   155→          animate={{ opacity: 1, y: 0 }}
   156→          className="text-center px-4"
   157→        >
   158→          <div
   159→            className="w-28 h-28 rounded-full mx-auto mb-6 flex items-center justify-center"
   160→            style={{ backgroundColor: '#f0f4f8' }}
   161→          >
   162→            <GitCompareArrows className="w-14 h-14" style={{ color: '#1E3A5F', opacity: 0.4 }} />
   163→          </div>
   164→          <h2 className="text-xl font-bold mb-2" style={{ color: '#1E3A5F' }}>
   165→            {t('compare.empty')}
   166→          </h2>
   167→          <p className="text-gray-500 mb-6">{t('compare.selectMin')}</p>
   168→          <Button
   169→            onClick={() => navigate('/catalogue')}
   170→            className="font-semibold shadow-md"
   171→            style={{ backgroundColor: '#1E3A5F', color: '#fff' }}
   172→          >
   173→            {t('compare.backToCatalogue')}
   174→          </Button>
   175→        </motion.div>
   176→      </div>
   177→    );
   178→  }
   179→

   183→      className="inline-flex items-center justify-center w-4 h-4 rounded-full text-[9px] font-bold ml-1 shrink-0"

   194→    getValue: (p: Product) => React.ReactNode;
   195→    isDifferent?: boolean;

   197→
   198→  const rows: RowDef[] = [
   199→    {
   200→      key: 'image',
   201→      label: t('compare.image'),
   202→      getValue: (p) => (
   203→        <div className="w-full aspect-square max-w-[160px] mx-auto rounded-xl overflow-hidden bg-white shadow-sm border border-gray-100">
   204→          {p.images && p.images.length > 0 ? (
   205→            <img
   206→              src={p.images[0]}
   207→              alt={getProductName(p, language)}
   208→              loading="lazy"
   209→              className="w-full h-full object-cover"
   210→            />
   211→          ) : (
   212→            <div
   213→              className="w-full h-full flex items-center justify-center"

   215→                background: 'linear-gradient(135deg, #1E3A5F 0%, #2A5F8F 60%, #3A7CB8 100%)',

   217→            >
   218→              <Package className="w-12 h-12 text-white/30" />
   219→            </div>

   285→          <span className="text-gray-400 text-sm">—</span>

   288→      isDifferent: isAttributeDifferent((p) => p.category_id),
   289→    },
   290→    {
   291→      key: 'stock',
   292→      label: t('compare.stock'),
   293→      getValue: (p) =>
   294→        p.in_stock ? (
   295→          <div className="flex items-center justify-center gap-1.5">
   296→            <CircleDot className="w-3.5 h-3.5 text-green-600" />
   297→            <span className="text-sm font-medium text-green-700">{t('products.inStock')}</span>
   298→          </div>
   299→        ) : (
   300→          <div className="flex items-center justify-center gap-1.5">
   301→            <XCircle className="w-3.5 h-3.5 text-red-500" />
   302→            <span className="text-sm font-medium text-red-600">{t('products.outOfStock')}</span>
   303→          </div>
   304→        ),
   305→      isDifferent: isAttributeDifferent((p) => String(p.in_stock)),
   306→    },
   307→    {
   308→      key: 'warranty',
   309→      label: t('compare.warranty'),
   310→      getValue: (p) =>
   311→        p.warranty && p.warranty !== '-' ? (
   312→          <div className="flex items-center justify-center gap-1.5">
   313→            <Shield className="w-3.5 h-3.5 text-blue-600" />
   314→            <span className="text-sm font-medium">{p.warranty}</span>
   315→          </div>
   316→        ) : (
   317→          <span className="text-gray-400 text-sm">—</span>
   318→        ),
   319→      isDifferent: isAttributeDifferent((p) => p.warranty || ''),
   320→    },
   321→    {
   322→      key: 'origin',
   323→      label: t('compare.origin'),
   324→      getValue: (p) =>
   325→        p.origin ? (
   326→          <div className="flex items-center justify-center gap-1.5">
   327→            <MapPin className="w-3.5 h-3.5" style={{ color: '#1E3A5F' }} />
   328→            <span className="text-sm">{p.origin}</span>
   329→          </div>
   330→        ) : (
   331→          <span className="text-gray-400 text-sm">—</span>
   332→        ),
   333→      isDifferent: isAttributeDifferent((p) => p.origin || ''),
   334→    },
   335→    {
   336→      key: 'ref',
   337→      label: t('compare.ref'),
   338→      getValue: (p) =>
   339→        p.ref ? (
   340→          <div className="flex items-center justify-center gap-1.5">
   341→            <Hash className="w-3.5 h-3.5 text-gray-400" />
   342→            <span className="text-xs font-mono">{p.ref}</span>
   343→          </div>
   344→        ) : (
   345→          <span className="text-gray-400 text-sm">—</span>
   346→        ),
   347→      isDifferent: isAttributeDifferent((p) => p.ref || ''),
   348→    },
   349→  ];
   350→
   351→  return (
   352→    <div className="min-h-screen bg-gray-50 pt-20">
   353→      {/* Header */}
   354→      <div
   355→        className="relative overflow-hidden"
   356→        style={{
   357→          background: 'linear-gradient(135deg, #1E3A5F 0%, #2A5F8F 50%, #1E3A5F 100%)',
   358→        }}
   359→      >
   360→        <div className="absolute inset-0 overflow-hidden pointer-events-none">
   361→          <div
   362→            className="absolute -top-24 -right-24 w-64 h-64 rounded-full opacity-10"
   363→            style={{ backgroundColor: '#F0C05A' }}
   364→          />
   365→          <div
   366→            className="absolute -bottom-16 -left-16 w-48 h-48 rounded-full opacity-10"
   367→            style={{ backgroundColor: '#F0C05A' }}
   368→          />
   369→        </div>
   370→
   371→        <div className="relative max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8 sm:py-10">
   372→          <motion.div
   373→            initial={{ opacity: 0, y: 20 }}
   374→            animate={{ opacity: 1, y: 0 }}
   375→            className="text-center"
   376→          >
   377→            <div className="flex items-center justify-center gap-3 mb-3">
   378→              <GitCompareArrows className="w-8 h-8" style={{ color: '#F0C05A' }} />
   379→              <h1 className="text-3xl sm:text-4xl font-bold text-white">
   380→                {t('compare.title')}
   381→              </h1>
   382→            </div>
   383→            <p className="text-blue-200 text-sm">
   384→              {compareProducts.length} {t('compare.items')}
   385→            </p>
   386→            {/* Legend for difference highlighting */}
   387→            <div className="mt-3 flex items-center justify-center gap-4 text-xs text-blue-200/70">
   388→              <span className="flex items-center gap-1.5">
   389→                <span className="w-3 h-3 rounded-sm" style={{ backgroundColor: 'rgba(240,192,90,0.15)' }} />
   390→                {language === 'ar' ? 'قيم مختلفة' : 'Valeurs différentes'}
   391→              </span>
   392→              <span className="flex items-center gap-1">
   393→                <span
   394→                  className="inline-flex items-center justify-center w-3.5 h-3.5 rounded-full text-[7px] font-bold"
   395→                  style={{ backgroundColor: 'rgba(240,192,90,0.2)', color: '#D4A017' }}
   396→                >
   397→                  ≠
   398→                </span>
   399→                {language === 'ar' ? 'مختلف' : 'Diffère'}
   400→              </span>
   401→            </div>
   402→          </motion.div>
   403→        </div>
   404→      </div>
   405→
   406→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6">
   407→        {/* Back button + Clear */}
   408→        <div className="flex items-center justify-between mb-6">
   409→          <Button
   410→            variant="outline"
   411→            onClick={() => navigate('/catalogue')}
   412→            className="gap-2 shadow-sm"
   413→            style={{ borderColor: '#1E3A5F', color: '#1E3A5F' }}
   414→          >
   415→            <ArrowLeft className="w-4 h-4" />
   416→            {t('compare.backToCatalogue')}
   417→          </Button>
   418→          <Button
   419→            variant="ghost"
   420→            onClick={clearCompare}
   421→            className="gap-2 text-red-500 hover:text-red-700 hover:bg-red-50"
   422→          >
   423→            <GitCompareArrows className="w-4 h-4" />
   424→            {t('compare.clear')}
   425→          </Button>
   426→        </div>
   427→
   428→        {/* Comparison Table */}
   429→        <motion.div
   430→          initial={{ opacity: 0, y: 20 }}
   431→          animate={{ opacity: 1, y: 0 }}
   432→          transition={{ delay: 0.1 }}
   433→        >
   434→          <Card className="border-0 shadow-sm overflow-hidden">
   435→            <CardContent className="p-0">
   436→              <div className="overflow-x-auto relative">
   437→                {/* Gradient fade indicators for horizontal scroll on mobile */}
   438→                <div className="sticky left-0 top-0 bottom-0 w-6 z-20 pointer-events-none absolute bg-gradient-to-r from-white to-transparent sm:hidden" />
   439→                <table className="w-full border-collapse">
   440→                  <thead>
   441→                    <tr className="border-b" style={{ backgroundColor: '#f8fafc' }}>
   442→                      <th
   443→                        className="p-4 text-left text-sm font-semibold min-w-[140px] sticky left-0 z-10 border-r border-gray-100"
   444→                        style={{
   445→                          backgroundColor: '#f8fafc',
   446→                          color: '#1E3A5F',
   447→                        }}
   448→                      >
   449→                        <div className="flex items-center gap-2">
   450→                          <GitCompareArrows className="w-4 h-4" style={{ color: '#F0C05A' }} />
   451→                          {t('compare.highlightDiff')}
   452→                        </div>
   453→                      </th>
   454→                      {compareProducts.map((product) => (
   455→                        <th key={product.id} className="p-4 min-w-[200px]">
   456→                          <div className="flex flex-col items-center">
   457→                            {/* Product image in header */}
   458→                            <div className="w-16 h-16 rounded-xl overflow-hidden mb-2 shadow-sm border border-gray-100 bg-white">
   459→                              {product.images && product.images.length > 0 ? (
   460→                                <img
   461→                                  src={product.images[0]}
   462→                                  alt={getProductName(product, language)}
   463→                                  loading="lazy"
   464→                                  className="w-full h-full object-cover"
   465→                                />
   466→                              ) : (

   468→                                  className="w-full h-full flex items-center justify-center"
   469→                                  style={{ background: 'linear-gradient(135deg, #1E3A5F, #2A5F8F)' }}

   471→                                  <Package className="w-5 h-5 text-white/40" />

   485→                    </tr>
   486→                  </thead>
   487→                  <tbody>
   488→                    {rows.map((row) => (

   493→                            ? ''
   494→                            : ''
   495→                        }`}
   496→                      >
   497→                        <td
   498→                          className={`p-4 text-sm font-medium sticky left-0 z-10 border-r border-gray-100 ${
   499→                            row.isDifferent ? '' : ''
   500→                          }`}
   501→                          style={{

   503→                            backgroundColor: row.isDifferent ? '#FFFBEB' : '#f8fafc',

   539→                    {/* Spec rows */}

   686→// Simple hash function for deterministic rating fallback
   687→function hashCode(str: string): number {
   688→  let hash = 0;
   689→  for (let i = 0; i < str.length; i++) {
   690→    const char = str.charCodeAt(i);
   691→    hash = ((hash << 5) - hash) + char;
   692→    hash |= 0;
   693→  }
   694→  return hash;
   695→}
