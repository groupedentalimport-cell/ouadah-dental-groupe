    33→    33→// Animated counter hook
    34→    34→function useCounter(end: number, duration = 2000) {
    35→    35→  const [count, setCount] = useState(0);
    36→    36→  const ref = useRef<HTMLDivElement>(null);
    37→    37→  const inView = useInView(ref, { once: true });
    38→    38→
    39→    39→  useEffect(() => {
    40→    40→    if (!inView) return;
    41→    41→    let start = 0;
    42→    42→    const step = end / (duration / 16);
    43→    43→    const timer = setInterval(() => {
    44→    44→      start += step;
    45→    45→      if (start >= end) {
    46→    46→        setCount(end);
    47→    47→        clearInterval(timer);
    48→    48→      } else {
    49→    49→        setCount(Math.floor(start));
    50→    50→      }
    51→    51→    }, 16);
    52→    52→    return () => clearInterval(timer);
    53→    53→  }, [inView, end, duration]);
    54→    54→
    55→    55→  return { count, ref };
    56→    56→}
