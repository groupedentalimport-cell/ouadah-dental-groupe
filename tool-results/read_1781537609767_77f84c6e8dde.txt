   499→interface DataContextType {
   500→  categories: Category[];
   501→  products: Product[];
   502→  blogPosts: BlogPost[];
   503→  contactInfo: ContactInfo;
   504→  loading: boolean;
   505→  error: string | null;
   506→  refresh: () => void;
   507→  getCategoryName: (cat: Category, lang: Language) => string;
   508→  getProductName: (product: Product, lang: Language) => string;
   509→  getProductDescription: (product: Product, lang: Language) => string;
   510→  getProductShortDescription: (product: Product, lang: Language) => string;
   511→}
   512→
   513→const DataContext = createContext<DataContextType>({
   514→  categories: defaultCategories,
   515→  products: defaultProducts,
   516→  blogPosts: defaultBlogPosts,
   517→  contactInfo: defaultContactInfo,
   518→  loading: false,
   519→  error: null,
   520→  refresh: () => {},
   521→  getCategoryName: () => '',
   522→  getProductName: () => '',
   523→  getProductDescription: () => '',
   524→  getProductShortDescription: () => '',
   525→});
   526→
   527→export function DataProvider({ children }: { children: React.ReactNode }) {
   528→  const [categories, setCategories] = useState<Category[]>(defaultCategories);
   529→  const [products, setProducts] = useState<Product[]>(defaultProducts);
   530→  const [blogPosts, setBlogPosts] = useState<BlogPost[]>(defaultBlogPosts);
   531→  const [contactInfo] = useState<ContactInfo>(defaultContactInfo);
   532→  const [loading, setLoading] = useState(true);
   533→  const [error, setError] = useState<string | null>(null);
   534→
   535→  const fetchData = useCallback(async () => {
   536→    if (!isSupabaseConfigured || !supabase) {
   537→      // Use default data when Supabase is not configured
   538→      setLoading(false);
   539→      return;
   540→    }
   541→
   542→    setLoading(true);
   543→    setError(null);
   544→
   545→    try {
   546→      // Fetch categories - map Supabase column names to our interface
   547→      const { data: catsData, error: catsError } = await supabase
   548→        .from('categories')
   549→        .select('*')
   550→        .order('ordre', { ascending: true });
   551→
   552→      if (!catsError && catsData && catsData.length > 0) {
   553→        const mapped = catsData.map((c: Record<string, unknown>) => ({
   554→          id: c.id as string,
   555→          slug: c.slug as string,
   556→          name_fr: (c.nom_fr || c.name_fr || '') as string,
   557→          name_ar: (c.nom_ar || c.name_ar || '') as string,
   558→          description_fr: (c.description_fr || '') as string,
   559→          description_ar: (c.description_ar || '') as string,
   560→          icon: (c.icone || c.icon || 'Package') as string,
   561→          image: (c.image || `/categories/${c.slug}.jpg`) as string,
   562→          sort_order: (c.ordre || c.sort_order || 0) as number,
   563→        }));
   564→        setCategories(mapped);
   565→      }
   566→
   567→      // Fetch products - map Supabase column names to our interface
   568→      const { data: prodsData, error: prodsError } = await supabase
   569→        .from('products')
   570→        .select('*')
   571→        .order('created_at', { ascending: false });
   572→
   573→      if (!prodsError && prodsData && prodsData.length > 0) {
   574→        const mapped = prodsData.map((p: Record<string, unknown>) => ({
   575→          id: p.id as string,
   576→          slug: p.slug as string,
   577→          name_fr: (p.nom_fr || p.name_fr || '') as string,
   578→          name_ar: (p.nom_ar || p.name_ar || '') as string,
   579→          description_fr: (p.description_fr || '') as string,
   580→          description_ar: (p.description_ar || '') as string,
   581→          short_description_fr: (p.short_description_fr || (p.description_fr as string || '').substring(0, 120)) as string,
   582→          short_description_ar: (p.short_description_ar || (p.description_ar as string || '').substring(0, 120)) as string,
   583→          category_id: p.category_id as string,
   584→          price: 0 as number, // Price not stored in Supabase - set on admin
   585→          images: ((p.images || []) as string[]).map((img: string) => {
   586→            // If already a full URL, return as-is
   587→            if (img.startsWith('http')) return img;
   588→            // Otherwise prefix with Supabase storage URL
   589→            const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
   590→            return `${supabaseUrl}/storage/v1/object/public/product-images/${img}`;
   591→          }),
   592→          specs: (p.specs || []) as Record<string, string>[],
   593→          features_fr: (p.features_fr || []) as string[],
   594→          features_ar: (p.features_ar || []) as string[],
   595→          in_stock: (p.disponible ?? p.in_stock ?? true) as boolean,
   596→          is_featured: (p.en_vedette ?? p.is_featured ?? false) as boolean,
   597→          is_new: (p.is_new ?? false) as boolean,
   598→          is_best_seller: (p.is_best_seller ?? false) as boolean,
   599→          brand: (p.marque || p.brand || '') as string,
   600→          origin: (p.origin || '') as string,
   601→          warranty: (p.warranty || '') as string,
   602→          delivery_time_fr: (p.delivery_time_fr || '') as string,
   603→          delivery_time_ar: (p.delivery_time_ar || '') as string,
   604→          installation_included: (p.installation_included ?? false) as boolean,
   605→          training_included: (p.training_included ?? false) as boolean,
   606→          ref: (p.ref || p.modele || '') as string,
   607→          target: (p.cible || p.target || []) as string[],
   608→          pdf_url: (p.pdf_url || '') as string,
   609→          brochure_pdf: (p.brochure_pdf || '') as string,
   610→          videos: (p.videos || []) as { title_fr: string; title_ar: string; url: string; type: string }[],
   611→          created_at: p.created_at as string,
   612→        }));
   613→        setProducts(mapped);
   614→      }
   615→
   616→      // Fetch blog posts - map Supabase column names
   617→      const { data: blogData, error: blogError } = await supabase
   618→        .from('blog_posts')
   619→        .select('*')
   620→        .order('created_at', { ascending: false });
   621→
   622→      if (!blogError && blogData && blogData.length > 0) {
   623→        const mapped = blogData.map((b: Record<string, unknown>) => ({
   624→          id: b.id as string,
   625→          slug: b.slug as string,
   626→          title_fr: (b.titre_fr || b.title_fr || '') as string,
   627→          title_ar: (b.titre_ar || b.title_ar || '') as string,
   628→          excerpt_fr: (b.excerpt_fr || '') as string,
   629→          excerpt_ar: (b.excerpt_ar || '') as string,
   630→          content_fr: (b.content_fr || '') as string,
   631→          content_ar: (b.content_ar || '') as string,
   632→          image: (() => {
   633→            const img = (b.image || '') as string;
   634→            if (!img || img.startsWith('http')) return img;
   635→            const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
   636→            return `${supabaseUrl}/storage/v1/object/public/blog-images/${img}`;
   637→          })(),
   638→          author: (b.author || 'ODG Team') as string,
   639→          date: (b.date || b.created_at || '') as string,
   640→          read_time: (b.read_time || 5) as number,
   641→          tags: (b.tags || []) as string[],
   642→          category: (b.category || '') as string,
   643→          created_at: b.created_at as string,
   644→        }));
   645→        setBlogPosts(mapped);
   646→      }
   647→    } catch (err) {
   648→      console.error('Error fetching data:', err);
   649→      setError('Failed to load data');
   650→    } finally {
   651→      setLoading(false);
   652→    }
   653→  }, []);
   654→
   655→  useEffect(() => {
   656→    fetchData();
   657→  }, [fetchData]);
   658→
   659→  const getCategoryName = useCallback((cat: Category, lang: Language): string => {
   660→    return lang === 'ar' ? cat.name_ar : cat.name_fr;
   661→  }, []);
   662→
   663→  const getProductName = useCallback((product: Product, lang: Language): string => {
   664→    return lang === 'ar' ? product.name_ar : product.name_fr;
   665→  }, []);
   666→
   667→  const getProductDescription = useCallback((product: Product, lang: Language): string => {
   668→    return lang === 'ar' ? product.description_ar : product.description_fr;
   669→  }, []);
   670→
   671→  const getProductShortDescription = useCallback((product: Product, lang: Language): string => {
   672→    return lang === 'ar' ? product.short_description_ar : product.short_description_fr;
   673→  }, []);
   674→
   675→  return (
   676→    <DataContext.Provider
   677→      value={{
   678→        categories,
   679→        products,
   680→        blogPosts,
   681→        contactInfo,
   682→        loading,
   683→        error,
   684→        refresh: fetchData,
   685→        getCategoryName,
   686→        getProductName,
   687→        getProductDescription,
   688→        getProductShortDescription,
   689→      }}
   690→    >
   691→      {children}
   692→    </DataContext.Provider>
   693→  );
   694→}
   695→
   696→export function useData() {
   697→  return useContext(DataContext);
   698→}
